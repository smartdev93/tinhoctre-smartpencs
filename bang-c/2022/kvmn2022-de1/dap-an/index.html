    Đáp án - Bảng C Khu vực miền Nam 2022¶

    Lời giải tham khảo bằng Python 3 và C++ (C++17). Xem lại đề bài.

    Đề chính thức đọc từ bàn phím và in ra màn hình. Để đồng bộ với các đề khác trên site, mỗi chương trình đọc input.inp và ghi output.out nếu có file input.inp; nếu không có file thì đọc từ bàn phím và in ra màn hình như đề yêu cầu.

    STT Bài Bảng Ý tưởng Độ phức tạp
    1 Bộ ba C1, C2 Với mỗi \(x\), đếm \(y\) trong một đoạn; gom các \(x\) có cùng \(\lfloor b_3/x \rfloor\) và \(\lceil a_3/x \rceil\) \(O(\sqrt{a_3} + \sqrt{b_3})\)
    2 Chọn cặp C1, C2 Tìm nhị phân ngưỡng \(T\) (trọng số của cặp thứ \(k\)), đếm và cộng các cặp \(\ge T\) bằng cây Fenwick \(O(n \log n \log V)\)
    3 Chọn nhóm C1 \(n \le 20\): vét cạn bitmask (chính xác). \(n \le 200\): tìm kiếm cục bộ lặp (heuristic) \(O(2^n)\) / \(O(\text{lần lặp} \cdot n)\)
    4 Xe buýt C2 Nếu \(b < a\): thử 2 trạm đặc biệt kề \(x\) và 2 trạm kề \(y\) \(O(n + q)\)

    1. Bộ ba¶

    Ý tưởng. Với mỗi \(x\) cố định thì \(z = xy\), nên chỉ cần đếm \(y\):

    • \(x = 0\): \(z = 0\), cần \(a_3 = 0\); khi đó mọi \(y \in [a_2, b_2]\) đều được.
    • \(x \ge 1\): \(a_3 \le xy \le b_3 \iff \lceil a_3/x \rceil \le y \le \lfloor b_3/x \rfloor\). Số \(y\) hợp lệ là độ dài đoạn \([\max(a_2, \lceil a_3/x \rceil),\ \min(b_2, \lfloor b_3/x \rfloor)]\) (0 nếu đoạn rỗng).

    Duyệt từng \(x\) tới \(10^9\) thì quá chậm. Nhưng \(\lfloor b_3/x \rfloor\) chỉ nhận khoảng \(2\sqrt{b_3}\) giá trị khác nhau, và các \(x\) cùng giá trị nằm liền nhau: với \(f = \lfloor b_3/x \rfloor > 0\), khối kéo dài tới \(x' = \lfloor b_3/f \rfloor\). Tương tự \(\lceil a_3/x \rceil = \lfloor (a_3-1)/x \rfloor + 1\). Trong một khối mà cả hai giá trị không đổi, số \(y\) hợp lệ cũng không đổi, nên nhân với độ dài khối. Tổng số khối khoảng \(2(\sqrt{a_3} + \sqrt{b_3}) \approx 1.3 \times 10^5\).

    Ví dụ 6 8 4 5 27 35: \(x = 6\): \(y \in [\max(4, 5), \min(5, 5)] = \{5\}\); \(x = 7\): \(y \in [4, 5]\); \(x = 8\): \(y \in [4, 4]\). Tổng \(1 + 2 + 1 = 4\).

    Bẫy thường gặp

    • Quên trường hợp \(x = 0\) (hoặc chia cho 0). Khi \(a_1 = 0\), \(a_3 = 0\) thì mọi \(y\) đều ghép được với \(x = 0\).
    • Quên trường hợp \(y = 0\): khi \(x > b_3\) thì \(\lfloor b_3/x \rfloor = 0\), vẫn đếm được \(y = 0\) nếu \(a_2 = a_3 = 0\). Công thức đoạn ở trên tự xử lý được.
    • Subtask \(a_1 = b_1\) chỉ cần một phép tính, nhưng lời giải chia khối đã bao luôn.

    Nếu viết bằng C++

    Đáp án tới khoảng \((10^9 + 1)^2 \approx 10^{18}\), vẫn vừa long long (tối đa khoảng \(9.2 \times 10^{18}\)). Tích (hi - lo + 1) * (end - x + 1) cũng phải tính bằng long long.

    boba.py
    import sys


def solve_boba(data):
    a1, b1, a2, b2, a3, b3 = map(int, data.split()[:6])
    ans = 0

    # x = 0 thì z = 0: cần a3 = 0, khi đó mọi y trong [a2, b2] đều được
    if a1 == 0:
        if a3 == 0 and a2 <= b2:
            ans += b2 - a2 + 1
        a1 = 1

    # x >= 1: y thuộc [max(a2, ceil(a3/x)), min(b2, b3 // x)]
    # Chia x thành các khối mà b3 // x và ceil(a3/x) không đổi (mỗi cái O(sqrt) giá trị)
    x = a1
    while x <= b1:
        f = b3 // x
        end = b1
        if f > 0:
            end = min(end, b3 // f)
        if a3 == 0:
            c = 0
        else:
            g = (a3 - 1) // x  # ceil(a3/x) = g + 1
            c = g + 1
            if g > 0:
                end = min(end, (a3 - 1) // g)
        lo = max(a2, c)
        hi = min(b2, f)
        if hi >= lo:
            ans += (hi - lo + 1) * (end - x + 1)
        x = end + 1
    return str(ans)


if __name__ == "__main__":
    try:
        with open("input.inp") as f:
            data = f.read()
        with open("output.out", "w") as f:
            f.write(solve_boba(data) + "\n")
    except FileNotFoundError:
        print(solve_boba(sys.stdin.read()))
    boba.cpp
    #include <algorithm>
#include <cstdio>
#include <iostream>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    // Có input.inp thì đọc/ghi file, không thì dùng bàn phím/màn hình
    if (FILE *f = fopen("input.inp", "r")) {
        fclose(f);
        freopen("input.inp", "r", stdin);
        freopen("output.out", "w", stdout);
    }
    long long a1, b1, a2, b2, a3, b3;
    cin >> a1 >> b1 >> a2 >> b2 >> a3 >> b3;
    // Kết quả tới khoảng 10^18 -> long long (vẫn dưới 9.2 * 10^18)
    long long ans = 0;

    // x = 0 thì z = 0: cần a3 = 0, khi đó mọi y trong [a2, b2] đều được
    if (a1 == 0) {
        if (a3 == 0 && a2 <= b2) ans += b2 - a2 + 1;
        a1 = 1;
    }

    // x >= 1: y thuộc [max(a2, ceil(a3/x)), min(b2, b3 / x)]
    // Chia x thành các khối mà b3 / x và ceil(a3/x) không đổi
    long long x = a1;
    while (x <= b1) {
        long long f = b3 / x;
        long long end = b1;
        if (f > 0) end = min(end, b3 / f);
        long long c = 0;
        if (a3 > 0) {
            long long g = (a3 - 1) / x;  // ceil(a3/x) = g + 1
            c = g + 1;
            if (g > 0) end = min(end, (a3 - 1) / g);
        }
        long long lo = max(a2, c), hi = min(b2, f);
        if (hi >= lo) ans += (hi - lo + 1) * (end - x + 1);
        x = end + 1;
    }
    cout << ans << "\n";
    return 0;
}

    2. Chọn cặp¶

    Ý tưởng. Đặt tổng tiền tố \(P_0 = 0\), \(P_i = a_1 + \cdots + a_i\). Cặp \((l, r)\) ứng với cặp chỉ số \((i, j) = (l - 1, r)\), có trọng số \(P_j - P_i\) và điều kiện độ dài thành \(j - i \ge X\). Bài toán thành: lấy tổng \(k\) giá trị \(P_j - P_i\) lớn nhất trong các cặp \(j - i \ge X\).

    Vì \(k\) tới \(10^7\), không lấy từng cặp ra được. Thay vào đó tìm ngưỡng \(T\) = trọng số của cặp lớn thứ \(k\):

    1. Hàm \(\text{cnt}(T)\) = số cặp có trọng số \(\ge T\). Duyệt \(j\) từ \(X\) tới \(n\), đưa \(P_{j-X}\) vào cây Fenwick (theo thứ hạng giá trị), rồi đếm số \(P_i \le P_j - T\) đã có trong cây. Mất \(O(n \log n)\).
    2. \(\text{cnt}(T)\) giảm khi \(T\) tăng, nên tìm nhị phân \(T\) lớn nhất có \(\text{cnt}(T) \ge k\). Trọng số nằm trong \([-3 \times 10^{10}, 3 \times 10^{10}]\), cần khoảng 37 lần.
    3. Tính thêm tổng trọng số các cặp \(> T\) (cây Fenwick thứ hai lưu tổng \(P_i\); mỗi \(j\) góp \(c \cdot P_j - \sum P_i\)). Các cặp \(> T\) đều được chọn, số cặp còn thiếu là \(k - \text{cnt}(T+1)\), và chúng đều có trọng số đúng bằng \(T\).

    Đáp án \(= \text{tổng}(> T) + (k - \text{cnt}(T + 1)) \cdot T\).

    Ví dụ: \(P = (0, 3, 5, -1, 7)\), \(X = 2\). Các trọng số: \(7, 5, 4, 2, -1, -4\). Với \(k = 4\), ngưỡng \(T = 2\); các cặp \(> 2\) là \(7, 5, 4\) (tổng 16), thiếu 1 cặp bằng 2. Đáp án \(16 + 2 = 18\).

    Bẫy thường gặp

    • Dùng hàng đợi ưu tiên lấy từng cặp (kiểu bài "Super Piano") chạy \(O((n + k) \log n)\): được các subtask \(k \le 3 \times 10^5\), nhưng chậm với \(k = 10^7\).
    • Nhiều cặp có cùng trọng số \(T\): không được cộng tất cả các cặp \(\ge T\), phải cộng đúng \(k - \text{cnt}(T+1)\) lần \(T\).
    • Điều kiện độ dài: cặp \((l, r)\) hợp lệ khi \(r - (l - 1) \ge X\), tức là khi xét \(j\) thì chỉ đưa vào cây các \(P_i\) với \(i \le j - X\).

    Nếu viết bằng C++

    Trọng số tới \(3 \times 10^{10}\) và đáp án tới \(10^7 \cdot 3 \times 10^{10} = 3 \times 10^{17}\): dùng long long cho tổng tiền tố, cây Fenwick lưu tổng, và biến đếm. Lời giải C++ chạy các test lớn nhất (\(n = 3 \times 10^5\)) trong khoảng 0.8 đến 1 giây trên máy thử.

    Về tốc độ bản Python

    Bản Python cùng thuật toán chạy khoảng 2 giây với \(n = 5 \times 10^4\), nhưng khoảng 20 giây với \(n = 3 \times 10^5\). Như vậy bản Python chỉ kịp các subtask \(n \le 5 \times 10^4\); subtask lớn cần C++.

    choncap.py
    import sys
from bisect import bisect_right


def solve_choncap(data):
    t = data.split()
    n, k, X = int(t[0]), int(t[1]), int(t[2])
    a = [int(v) for v in t[3:3 + n]]

    # P[i] = a1 + ... + ai; trọng số cặp (l, r) = P[r] - P[l-1], cần r - (l-1) >= X
    P = [0] * (n + 1)
    for i in range(n):
        P[i + 1] = P[i] + a[i]
    vals = sorted(set(P))
    rank = [bisect_right(vals, v) for v in P]  # vị trí 1..len(vals) trên cây Fenwick
    sz = len(vals)

    total = (n - X + 1) * (n - X + 2) // 2  # số cặp hợp lệ
    k = min(k, total)  # (giả định) k không vượt quá số cặp

    def count(T, need_sum):
        # Đếm số cặp có trọng số >= T (và tổng trọng số của chúng nếu need_sum)
        bc = [0] * (sz + 1)
        bs = [0] * (sz + 1)
        cnt = 0
        sm = 0
        for j in range(X, n + 1):
            # đưa P[j - X] vào cây: các i = 0..j-X là đầu trái hợp lệ cho j
            i = rank[j - X]
            v = P[j - X]
            while i <= sz:
                bc[i] += 1
                if need_sum:
                    bs[i] += v
                i += i & -i
            # các P[i] <= P[j] - T
            i = bisect_right(vals, P[j] - T)
            c = 0
            s = 0
            while i > 0:
                c += bc[i]
                if need_sum:
                    s += bs[i]
                i -= i & -i
            cnt += c
            if need_sum:
                sm += c * P[j] - s
        return cnt, sm

    # Tìm ngưỡng T lớn nhất mà số cặp có trọng số >= T vẫn >= k
    abs_sum = sum(abs(v) for v in a)  # mọi trọng số nằm trong [-abs_sum, abs_sum]
    lo = -2 * abs_sum - 1  # count(lo) = total >= k
    hi = abs_sum + 1       # count(hi) = 0 < k
    while hi - lo > 1:
        mid = (lo + hi) // 2
        if count(mid, False)[0] >= k:
            lo = mid
        else:
            hi = mid
    T = lo
    # Lấy hết các cặp > T, phần còn thiếu đều có trọng số đúng bằng T
    c, s = count(T + 1, True)
    return str(s + (k - c) * T)


if __name__ == "__main__":
    try:
        with open("input.inp") as f:
            data = f.read()
        with open("output.out", "w") as f:
            f.write(solve_choncap(data) + "\n")
    except FileNotFoundError:
        print(solve_choncap(sys.stdin.read()))
    choncap.cpp
    #include <algorithm>
#include <cstdio>
#include <iostream>
#include <vector>
using namespace std;

int n, X, sz;
vector<long long> P, vals;
vector<int> rk;
vector<long long> bc, bs;

// Đếm số cặp có trọng số >= T, tổng trọng số của chúng ghi vào sm (nếu needSum)
long long countPairs(long long T, bool needSum, long long &sm) {
    fill(bc.begin(), bc.end(), 0);
    if (needSum) fill(bs.begin(), bs.end(), 0);
    long long cnt = 0;
    sm = 0;
    for (int j = X; j <= n; j++) {
        // đưa P[j - X] vào cây: các i = 0..j-X là đầu trái hợp lệ cho j
        long long v = P[j - X];
        for (int i = rk[j - X]; i <= sz; i += i & -i) {
            bc[i]++;
            if (needSum) bs[i] += v;
        }
        // các P[i] <= P[j] - T
        int i = upper_bound(vals.begin(), vals.end(), P[j] - T) - vals.begin();
        long long c = 0, s = 0;
        for (; i > 0; i -= i & -i) {
            c += bc[i];
            if (needSum) s += bs[i];
        }
        cnt += c;
        if (needSum) sm += c * P[j] - s;
    }
    return cnt;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    // Có input.inp thì đọc/ghi file, không thì dùng bàn phím/màn hình
    if (FILE *f = fopen("input.inp", "r")) {
        fclose(f);
        freopen("input.inp", "r", stdin);
        freopen("output.out", "w", stdout);
    }
    long long k;
    cin >> n >> k >> X;
    // P[i] = a1 + ... + ai; trọng số cặp (l, r) = P[r] - P[l-1], cần r - (l-1) >= X
    P.assign(n + 1, 0);
    long long absSum = 0;
    for (int i = 1; i <= n; i++) {
        long long x;
        cin >> x;
        P[i] = P[i - 1] + x;
        absSum += (x < 0 ? -x : x);
    }
    vals = P;
    sort(vals.begin(), vals.end());
    vals.erase(unique(vals.begin(), vals.end()), vals.end());
    sz = vals.size();
    rk.resize(n + 1);
    for (int i = 0; i <= n; i++) rk[i] = upper_bound(vals.begin(), vals.end(), P[i]) - vals.begin();
    bc.assign(sz + 1, 0);
    bs.assign(sz + 1, 0);

    long long total = (long long)(n - X + 1) * (n - X + 2) / 2;  // số cặp hợp lệ
    k = min(k, total);  // (giả định) k không vượt quá số cặp

    // Tìm ngưỡng T lớn nhất mà số cặp có trọng số >= T vẫn >= k
    long long lo = -2 * absSum - 1, hi = absSum + 1, sm;
    while (hi - lo > 1) {
        long long mid = lo + (hi - lo) / 2;
        if (countPairs(mid, false, sm) >= k) lo = mid;
        else hi = mid;
    }
    long long T = lo;
    // Lấy hết các cặp > T, phần còn thiếu đều có trọng số đúng bằng T
    long long c = countPairs(T + 1, true, sm);
    // Kết quả tới khoảng 10^7 * 3 * 10^10 = 3 * 10^17 -> long long
    cout << sm + (k - c) * T << "\n";
    return 0;
}

    3. Chọn nhóm¶

    Đây là bài tối ưu tổ hợp khó (tìm tập đỉnh có tổng trọng số cạnh bên trong lớn nhất, có cạnh âm; bài toán này NP-khó nói chung). Đề chấm theo tỉ lệ \(\min\{1, c^3/q^3\}\) so với đáp án của Ban giám khảo, nên không cần tối ưu tuyệt đối để có điểm. Lời giải tham khảo gồm hai phần.

    Phần 1, \(n \le 20\): vét cạn (chính xác). Gọi \(\text{val}[S]\) là tổng sự phối hợp của tập \(S\). Với \(v\) là học sinh có chỉ số lớn nhất trong \(S\) và \(R = S \setminus \{v\}\):

    \[\text{val}[S] = \text{val}[R] + \sum_{u \in R} c(v, u)\]

    Tổng \(\sum_{u \in R} c(v, u)\) tra bảng: tách \(R\) thành 10 bit thấp và 10 bit cao, mỗi phần lập sẵn bảng 1024 phần tử cho từng \(v\). Tổng cộng \(O(2^n + n \cdot 1024)\), khoảng \(10^6\) phép tính. In tập có tổng lớn nhất.

    Phần 2, \(n \le 200\): tìm kiếm cục bộ lặp (heuristic, KHÔNG đảm bảo tối ưu).

    • Giữ \(\text{gain}[v] = \sum_{u \in S} c(v, u)\). Đổi trạng thái \(v\) (thêm vào hoặc bỏ ra) làm tổng thay đổi \(+\text{gain}[v]\) hoặc \(-\text{gain}[v]\); cập nhật \(\text{gain}\) mất \(O(n)\).
    • Leo đồi: lặp lại việc đổi trạng thái học sinh làm tổng tăng nhiều nhất, tới khi không còn bước nào làm tăng.
    • Nhiễu: đổi trạng thái 1 tới 4 học sinh ngẫu nhiên rồi leo đồi lại; nếu kết quả tệ hơn trước khi nhiễu thì quay về trạng thái cũ. Cứ 500 lần thì khởi tạo lại từ một tập ngẫu nhiên. Tổng cộng 20000 lần, luôn ghi nhớ tập tốt nhất đã gặp.
    • Bộ sinh số ngẫu nhiên là LCG 64 bit với seed cố định, viết giống hệt nhau ở hai bản, nên chạy lại luôn ra cùng kết quả và bản Python, C++ in ra cùng một nhóm.

    Nhóm in ra luôn hợp lệ (chỉ số khác nhau, trong \([1, n]\)) và tổng in ra đúng là tổng của nhóm đó. Nếu không tìm được nhóm có tổng dương, lời giải in nhóm chỉ gồm học sinh 1 (tổng 0).

    Kết quả thử nghiệm của nhóm biên soạn: với 150 test ngẫu nhiên \(12 \le n \le 20\), heuristic (chỉ chạy 1000 lần lặp) đều tìm được đúng tối ưu của vét cạn. Với \(n = 200\), dạng test \(c \in \{1, -10^9\}\) trên đồ thị đầy đủ thì đáp án đúng là \(\binom{t}{2}\) với \(t\) là clique lớn nhất của các cặp \(+1\), nên so được với thuật toán tìm clique chính xác. Ở một test có 50% cặp \(+1\), lời giải đạt đúng tối ưu (clique 11, tổng 55). Ở một test có 60% cặp \(+1\), lời giải chỉ tìm được nhóm 13 người (tổng 78) trong khi tối ưu là 14 người (tổng 91), tỉ lệ \(78/91 \approx 0.86\), tức là chỉ được khoảng 63% điểm test đó. Vậy lời giải không tối ưu ở các test lớn và có thể mất điểm đáng kể.

    Ví dụ: các nhóm \(\{1, 2, 5\}\) và \(\{2, 5\}\) cùng có tổng 3 (lớn nhất). Lời giải tham khảo in 2 / 2 5, khác ví dụ của đề nhưng cùng tổng nên cùng điểm.

    Bẫy thường gặp

    • Chỉ thêm học sinh mà không bao giờ bỏ ra: dễ kẹt ở nghiệm kém khi có cạnh âm lớn.
    • Subtask \(c \in \{1, -10^9\}\): một cạnh \(-10^9\) làm tổng âm ngay, nên thực chất là chọn tập không chứa cặp "xung khắc" nào và có nhiều cặp \(+1\) nhất. Tìm kiếm cục bộ tự tránh các cặp này vì đổi trạng thái tạo ra cặp \(-10^9\) không bao giờ làm tăng tổng.
    • Điểm tính theo lập phương tỉ lệ: đạt 90% tổng của giám khảo chỉ được khoảng 73% điểm test.

    Nếu viết bằng C++

    Tổng tới \(m \cdot 10^9 \approx 2 \times 10^{13}\), dùng long long. Bản C++ chạy 20000 lần lặp với \(n = 200\) dưới 0.1 giây, bản Python khoảng 1.5 đến 2 giây. Với C++ có thể tăng số lần lặp (hoặc chạy theo thời gian) để kết quả tốt hơn, khi đó hai bản sẽ không còn in ra cùng nhóm.

    chonnhom.py
    import sys
from operator import add, sub

MASK64 = (1 << 64) - 1


def exact(n, W):
    # Vét cạn mọi tập con: val[mask] = val[mask bỏ bit cao nhất v] + tổng W[v][u] với u trong phần còn lại.
    # Tổng W[v][u] tra bảng theo 10 bit thấp (lo) và 10 bit cao (hi) của phần còn lại.
    val = [0]
    for v in range(n):
        lo = [0] * 1024
        hi = [0] * 1024
        for r in range(1, 1024):
            b = (r & -r).bit_length() - 1
            lo[r] = lo[r & (r - 1)] + (W[v][b] if b < n else 0)
            hi[r] = hi[r & (r - 1)] + (W[v][b + 10] if b + 10 < n else 0)
        val.extend([val[r] + lo[r & 1023] + hi[r >> 10] for r in range(1 << v)])
    best = max(range(len(val)), key=val.__getitem__)  # mask đầu tiên đạt max
    return val[best], [i + 1 for i in range(n) if best >> i & 1]


def heuristic(n, W, iters=20000):
    # Tìm kiếm cục bộ lặp (iterated local search), KHÔNG đảm bảo tối ưu.
    # Bộ sinh số ngẫu nhiên LCG 64 bit, seed cố định -> chạy lại cho cùng kết quả.
    state = [20220101]

    def rnd():
        state[0] = (state[0] * 6364136223846793005 + 1442695040888963407) & MASK64
        return state[0] >> 33

    ins = [0] * n   # ins[v] = 1 nếu v đang được chọn
    gain = [0] * n  # gain[v] = tổng W[v][u] với u đang được chọn
    cur = [0]       # tổng sự phối hợp của nhóm hiện tại

    def flip(v):
        if ins[v]:
            cur[0] -= gain[v]
            ins[v] = 0
            gain[:] = map(sub, gain, W[v])
        else:
            cur[0] += gain[v]
            ins[v] = 1
            gain[:] = map(add, gain, W[v])

    def climb():
        # Lặp: đổi trạng thái (chọn/bỏ) học sinh làm tăng tổng nhiều nhất, tới khi không tăng được
        while True:
            d = [-g if s else g for g, s in zip(gain, ins)]
            m = max(d)
            if m <= 0:
                return
            flip(d.index(m))

    climb()
    best = cur[0]
    best_set = ins[:]
    for it in range(iters):
        saved = (ins[:], gain[:], cur[0])
        restart = it % 500 == 499
        if restart:
            # khởi tạo lại từ một tập ngẫu nhiên
            for v in range(n):
                if rnd() & 1 != ins[v]:
                    flip(v)
        else:
            # nhiễu: đổi trạng thái 1..4 học sinh ngẫu nhiên
            for _ in range(1 + rnd() % 4):
                flip(rnd() % n)
        climb()
        if not restart and cur[0] < saved[2]:
            ins[:], gain[:], cur[0] = saved[0], saved[1], saved[2]  # tệ hơn thì quay lại
        if cur[0] > best:
            best = cur[0]
            best_set = ins[:]
    return best, [i + 1 for i in range(n) if best_set[i]]


def solve_chonnhom(data):
    t = data.split()
    n, m = int(t[0]), int(t[1])
    W = [[0] * n for _ in range(n)]
    for k in range(m):
        i = int(t[2 + 3 * k]) - 1
        j = int(t[3 + 3 * k]) - 1
        c = int(t[4 + 3 * k])
        if i != j:  # (giả định) bỏ qua cặp i = j; cặp lặp lại thì cộng dồn
            W[i][j] += c
            W[j][i] += c
    if n <= 20:
        value, group = exact(n, W)
    else:
        value, group = heuristic(n, W)
    if value <= 0:
        group = [1]  # không có nhóm tổng dương: chọn 1 học sinh (tổng 0)
    return str(len(group)) + "\n" + " ".join(map(str, group))

#include <cstdio>
#include <iostream>
#include <vector>
using namespace std;

typedef long long ll;
int n;
vector<vector<ll>> W;

// Vét cạn mọi tập con (n <= 20): val[mask] = val[mask bỏ bit cao nhất v] + tổng W[v][u], u thuộc phần còn lại
ll exact(vector<int> &group) {
    vector<ll> val(1u << n);
    val[0] = 0;
    vector<ll> lo(1024), hi(1024);
    for (int v = 0; v < n; v++) {
        lo[0] = hi[0] = 0;
        for (int r = 1; r < 1024; r++) {
            int b = __builtin_ctz(r);
            lo[r] = lo[r & (r - 1)] + (b < n ? W[v][b] : 0);
            hi[r] = hi[r & (r - 1)] + (b + 10 < n ? W[v][b + 10] : 0);
        }
        for (int r = 0; r < (1 << v); r++) val[(1 << v) | r] = val[r] + lo[r & 1023] + hi[r >> 10];
    }
    int best = 0;
    for (int mask = 1; mask < (1 << n); mask++)
        if (val[mask] > val[best]) best = mask;  // mask đầu tiên đạt max
    for (int i = 0; i < n; i++)
        if (best >> i & 1) group.push_back(i + 1);
    return val[best];
}

// Tìm kiếm cục bộ lặp (iterated local search), KHÔNG đảm bảo tối ưu.
// LCG 64 bit, seed cố định -> chạy lại cho cùng kết quả (và trùng bản Python)
uint64_t state = 20220101;
uint64_t rnd() {
    state = state * 6364136223846793005ULL + 1442695040888963407ULL;
    return state >> 33;
}
vector<int> ins;   // ins[v] = 1 nếu v đang được chọn
vector<ll> gain;   // gain[v] = tổng W[v][u] với u đang được chọn
ll cur = 0;        // tổng sự phối hợp của nhóm hiện tại

void flip(int v) {
    if (ins[v]) {
        cur -= gain[v];
        ins[v] = 0;
        for (int u = 0; u < n; u++) gain[u] -= W[v][u];
    } else {
        cur += gain[v];
        ins[v] = 1;
        for (int u = 0; u < n; u++) gain[u] += W[v][u];
    }
}

// Lặp: đổi trạng thái học sinh làm tăng tổng nhiều nhất, tới khi không tăng được
void climb() {
    while (true) {
        ll m = 0;
        int bv = -1;
        for (int v = 0; v < n; v++) {
            ll d = ins[v] ? -gain[v] : gain[v];
            if (bv < 0 || d > m) m = d, bv = v;
        }
        if (m <= 0) return;
        flip(bv);
    }
}

ll heuristic(vector<int> &group, int iters = 20000) {
    ins.assign(n, 0);
    gain.assign(n, 0);
    cur = 0;
    climb();
    ll best = cur;
    vector<int> bestSet = ins;
    for (int it = 0; it < iters; it++) {
        vector<int> sIns = ins;
        vector<ll> sGain = gain;
        ll sCur = cur;
        bool restart = it % 500 == 499;
        if (restart) {
            // khởi tạo lại từ một tập ngẫu nhiên
            for (int v = 0; v < n; v++)
                if ((int)(rnd() & 1) != ins[v]) flip(v);
        } else {
            // nhiễu: đổi trạng thái 1..4 học sinh ngẫu nhiên
            int cnt = 1 + rnd() % 4;
            for (int t = 0; t < cnt; t++) flip(rnd() % n);
        }
        climb();
        if (!restart && cur < sCur) ins = sIns, gain = sGain, cur = sCur;  // tệ hơn thì quay lại
        if (cur > best) best = cur, bestSet = ins;
    }
    for (int i = 0; i < n; i++)
        if (bestSet[i]) group.push_back(i + 1);
    return best;
}

    4. Xe buýt¶

    Ý tưởng. Đi xe thường giữa hai trạm \(u, v\) bất kỳ mất \(a|u - v|\), đi xe nhanh giữa hai trạm đặc biệt mất \(b|u - v|\).

    • Nếu \(b \ge a\): xe nhanh không bao giờ rẻ hơn, đáp án \(a|x - y|\).
    • Nếu \(b < a\): hành trình tốt nhất có dạng xe thường \(x \to s\), xe nhanh \(s \to u\), xe thường \(u \to y\) (\(s, u\) là trạm đặc biệt). Nhiều chặng xe nhanh liên tiếp gộp được thành một chặng (cùng giá \(b\) mỗi đoạn), còn chen chặng xe thường vào giữa chỉ đắt hơn.

    Với \(u\) cố định, chi phí \(a|x - s| + b|s - u|\) là hàm lồi theo \(s\) và nhỏ nhất tại \(s = x\) (vì \(a > b\)). Do đó trong các trạm đặc biệt, \(s\) tốt nhất là trạm đặc biệt gần nhất bên trái hoặc bên phải \(x\). Tương tự với \(u\) và \(y\). Chỉ cần thử \(2 \times 2 = 4\) tổ hợp, cộng với phương án đi toàn xe thường.

    Lập sẵn \(\text{pre}[i]\) (trạm đặc biệt gần nhất \(\le i\)) và \(\text{nxt}[i]\) (gần nhất \(\ge i\)) trong \(O(n)\), mỗi câu hỏi trả lời \(O(1)\).

    Ví dụ: câu hỏi \(1 \to 5\) với trạm đặc biệt \(\{2, 4\}\), \(a = 2\), \(b = 1\): \(s = 2\), \(u = 4\), chi phí \(2 \cdot 1 + 1 \cdot 2 + 2 \cdot 1 = 6\).

    Bẫy thường gặp

    • Chỉ thử trạm đặc biệt gần \(x\) nhất về một phía: có thể cần đi lùi một chút để lên xe nhanh.
    • Chạy Dijkstra cho từng câu hỏi chỉ được các subtask \(n, q \le 100\).
    • Các \(p_i\) không chắc đã sắp xếp hay khác nhau; đánh dấu bằng mảng thì không cần quan tâm.

    Nếu viết bằng C++

    Chi phí tới \(10^6 \cdot 10^6 = 10^{12}\), dùng long long. Dữ liệu tới \(3 \times 10^6\) số: bản C++ đọc bằng fread tự viết, vì cin của một số thư viện chuẩn (ví dụ libc++ trên macOS) mất khoảng 2.5 giây chỉ để đọc. Bản Python chạy test \(n = m = q = 10^6\) khoảng 2.3 giây.

    xebuyt.py
    import sys


def solve_xebuyt(data):
    t = data.split()
    n, m, q, a, b = (int(v) for v in t[:5])
    p = [int(v) for v in t[5:5 + m]]
    pos = 5 + m

    # pre[i] = trạm đặc biệt gần nhất <= i, nxt[i] = trạm đặc biệt gần nhất >= i (0 nếu không có)
    special = [False] * (n + 2)
    for v in p:
        special[v] = True
    pre = [0] * (n + 2)
    for i in range(1, n + 1):
        pre[i] = i if special[i] else pre[i - 1]
    nxt = [0] * (n + 2)
    for i in range(n, 0, -1):
        nxt[i] = i if special[i] else nxt[i + 1]

    out = []
    for _ in range(q):
        x = int(t[pos])
        y = int(t[pos + 1])
        pos += 2
        best = a * abs(x - y)  # chỉ đi xe thường
        if b < a:
            # xe thường từ x tới trạm đặc biệt s, xe nhanh s -> u, xe thường u -> y;
            # s chỉ cần thử 2 trạm đặc biệt kề x, u chỉ cần thử 2 trạm kề y
            for s in (pre[x], nxt[x]):
                if s == 0:
                    continue
                cs = a * abs(x - s)
                for u in (pre[y], nxt[y]):
                    if u == 0:
                        continue
                    c = cs + b * abs(s - u) + a * abs(u - y)
                    if c < best:
                        best = c
        out.append(best)
    return "\n".join(map(str, out))

static char buf[1 << 16];
size_t bufLen = 0, bufPos = 0;
int readChar() {
    if (bufPos == bufLen) {
        bufLen = fread(buf, 1, sizeof(buf), stdin);
        bufPos = 0;
        if (bufLen == 0) return -1;
    }
    return buf[bufPos++];
}
long long readInt() {
    int c = readChar();
    while (c != -1 && c != '-' && (c < '0' || c > '9')) c = readChar();
    bool neg = (c == '-');
    if (neg) c = readChar();
    long long x = 0;
    while (c >= '0' && c <= '9') x = x * 10 + (c - '0'), c = readChar();
    return neg ? -x : x;
}

int main() {
    // Có input.inp thì đọc/ghi file, không thì dùng bàn phím/màn hình

    for (int i = 0; i < m; i++) {

    }

    vector<int> pre(n + 2, 0), nxt(n + 2, 0);
    for (int i = 1; i <= n; i++) pre[i] = special[i] ? i : pre[i - 1];

        out += '\n';
    }
    fputs(out.c_str(), stdout);
    return 0;
}
