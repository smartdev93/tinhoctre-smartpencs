    Đáp án - Bảng C Vòng sơ khảo 2022 (26/06/2022)¶

    Lời giải tham khảo bằng Python 3 và C++ (C++17). Xem lại đề bài.

    Đề chính thức đọc từ bàn phím và in ra màn hình. Để đồng bộ với các đề khác trên site, mỗi chương trình đọc input.inp và ghi output.out nếu có file input.inp; nếu không có thì đọc từ bàn phím và in ra màn hình, đúng như đề yêu cầu.

    STT Bài Ý tưởng Độ phức tạp
    1 Biến đổi xâu Cây phân đoạn lười trên các khối 64 ký tự, nhãn là ánh xạ 26 chữ cái \(O(m \cdot (26 \log n + 64) + 26 \cdot n / 64 + n)\)
    2 Ma trận Thuật toán Freivalds: so sánh \(A(A(Av))\) với \(Bv\) cho vectơ ngẫu nhiên \(v\), tính theo mô-đun \(2^{61} - 1\) \(O(T \cdot n^2)\)
    3 Cắt ghép xâu Tham lam: lặp lại việc ghép đoạn chung dài nhất (băm xâu), thử 4 biến thể, lấy phương án tốt nhất khoảng \(O(n \log n)\) mỗi lượt độ dài
    4 Dãy cấp số nhân Quy hoạch động theo giá trị cuối và độ dài; dãy dài tối đa 30 phần tử \(O(30 \cdot n)\)

    1. Biến đổi xâu¶

    Ý tưởng. Thao tác \((i, j, c_1, c_2)\) có thể xem là áp một ánh xạ \(g\) trên 26 chữ cái lên mọi vị trí trong \([i, j]\): \(g(c_1) = c_2\), các chữ khác giữ nguyên. Áp lần lượt nhiều thao tác lên cùng một vị trí tương đương áp hợp các ánh xạ, và hợp hai ánh xạ chỉ tốn 26 phép gán: \(h(x) = g(f(x))\).

    Vì vậy ta dùng cây phân đoạn lười (lazy segment tree): mỗi nút giữ một ánh xạ "chưa đẩy xuống". Gán thao tác cho một đoạn thì chỉ cần hợp \(g\) vào \(O(\log n)\) nút phủ đoạn đó. Thứ tự quan trọng: thao tác ở nút cha luôn mới hơn thao tác đang chờ ở nút con, nên trước khi gán phải đẩy (push) nhãn của các nút tổ tiên xuống, rồi khi đẩy thì áp nhãn cha sau nhãn con. Cuối cùng đẩy hết nhãn xuống lá và áp vào xâu.

    Để giảm bộ nhớ và thời gian, lá của cây không phải từng ký tự mà là khối 64 ký tự. Khối chỉ bị phủ một phần ở hai đầu đoạn thì đẩy nhãn xuống khối đó, áp vào các ký tự, rồi sửa trực tiếp các ký tự trong đoạn (tối đa \(2 \cdot 64\) ký tự mỗi thao tác). Bản Python dùng bảng dịch 256 byte và bytes.translate để hợp ánh xạ và sửa cả khối bằng code C của thư viện chuẩn.

    Ví dụ: sau thao tác 0 2 a y, đoạn \([0, 2]\) mang ánh xạ \(a \to y\). Thao tác 1 3 a z trên \([1, 3]\) hợp \(a \to z\) sau đó, nhưng với vị trí 1, 2 thì ký tự lúc này là y (vì \(a \to y\) đã áp trước) nên không đổi, chỉ vị trí 3 đổi a thành z.

    Bẫy thường gặp

    • Duyệt thẳng từng vị trí trong đoạn tốn \(O(n \cdot m)\), tới \(1.5 \cdot 10^{11}\) phép tính với test lớn nhất. Chỉ qua được 2 subtask đầu.
    • Hợp ánh xạ sai thứ tự: thao tác mới phải áp sau thao tác cũ. Với \(g\): \(a \to b\) rồi \(h\): \(b \to c\), chữ a phải thành c; đảo thứ tự thì a chỉ thành b.
    • Quên đẩy nhãn của nút tổ tiên trước khi gán nhãn mới cho nút con làm đảo thứ tự hai thao tác.
    • Subtask chỉ có 'a'/'b' có lời giải riêng đơn giản hơn, nhưng lời giải tổng quát ở đây qua luôn subtask đó.

    Nếu viết bằng Python

    Trên máy kiểm thử, bản C++ chạy test lớn nhất (\(n = 10^6\), \(m = 1.5 \cdot 10^5\)) khoảng 0,55 giây, còn bản Python khoảng 3 giây, vượt giới hạn 1 giây. Bản Python đúng thuật toán nhưng có thể chỉ qua các subtask nhỏ.

    biendoixau.py
    import sys

B = 64                    # kích thước một khối ký tự (lá của cây phân đoạn)
ID = bytes(range(256))    # bảng dịch đồng nhất: không đổi ký tự nào


def solve_biendoixau(data):
    tokens = data.split()
    s = bytearray(tokens[0].encode())
    n = len(s)
    m = int(tokens[1])
    nb = (n + B - 1) // B
    size, log = 1, 0
    while size < nb:
        size *= 2
        log += 1
    # tag[k] = bảng dịch 256 byte "ký tự gốc -> ký tự hiện tại" chưa đẩy xuống của nút k
    tag = [ID] * (2 * size)

    def push(k):
        t = tag[k]
        if t is not ID:
            # thao tác ở nút cha mới hơn thao tác đang chờ ở nút con -> áp sau
            tag[2 * k] = tag[2 * k].translate(t)
            tag[2 * k + 1] = tag[2 * k + 1].translate(t)
            tag[k] = ID

    def materialize(blk):
        # đẩy hết thao tác đang chờ xuống khối blk rồi áp thẳng vào xâu
        k = blk + size
        for h in range(log, 0, -1):
            push(k >> h)
        t = tag[k]
        if t is not ID:
            a = blk * B
            b = min(a + B, n)
            s[a:b] = s[a:b].translate(t)
            tag[k] = ID

    tables = {}
    pos = 2
    for _ in range(m):
        i, j = int(tokens[pos]), int(tokens[pos + 1])
        c1, c2 = tokens[pos + 2], tokens[pos + 3]
        pos += 4
        g = tables.get((c1, c2))
        if g is None:
            tb = bytearray(ID)
            tb[ord(c1)] = ord(c2)
            g = tables[(c1, c2)] = bytes(tb)
        bi, bj = i // B, j // B
        if bi == bj:
            materialize(bi)
            s[i:j + 1] = s[i:j + 1].translate(g)
            continue
        lo, hi = bi, bj
        if i != bi * B:                      # khối trái chỉ phủ một phần
            materialize(bi)
            s[i:(bi + 1) * B] = s[i:(bi + 1) * B].translate(g)
            lo = bi + 1
        if j + 1 != min((bj + 1) * B, n):    # khối phải chỉ phủ một phần
            materialize(bj)
            s[bj * B:j + 1] = s[bj * B:j + 1].translate(g)
            hi = bj - 1
        if lo > hi:
            continue
        # gán thao tác cho các khối nguyên [lo, hi] (cây phân đoạn lười, kiểu từ dưới lên)
        l, r = lo + size, hi + 1 + size
        for h in range(log, 0, -1):
            if ((l >> h) << h) != l:
                push(l >> h)
            if ((r >> h) << h) != r:
                push((r - 1) >> h)
        while l < r:
            if l & 1:
                tag[l] = tag[l].translate(g)
                l += 1
            if r & 1:
                r -= 1
                tag[r] = tag[r].translate(g)
            l >>= 1
            r >>= 1
    # đẩy mọi thao tác xuống lá rồi áp vào từng khối
    for k in range(1, size):
        push(k)
    for blk in range(nb):
        t = tag[blk + size]
        if t is not ID:
            a = blk * B
            b = min(a + B, n)
            s[a:b] = s[a:b].translate(t)
    return s.decode()


if __name__ == "__main__":
    try:
        with open("input.inp") as f:
            data = f.read()
        with open("output.out", "w") as f:
            f.write(solve_biendoixau(data) + "\n")
    except FileNotFoundError:
        print(solve_biendoixau(sys.stdin.read()))
    biendoixau.cpp
    #include <algorithm>
#include <array>
#include <cstdio>
#include <iostream>
#include <string>
#include <vector>
using namespace std;

const int B = 64;  // kích thước một khối ký tự (lá của cây phân đoạn)
typedef array<unsigned char, 26> Map;  // ánh xạ "ký tự gốc -> ký tự hiện tại"

int n, sz, lg;
string s;
vector<Map> tag;      // thao tác chưa đẩy xuống của mỗi nút
vector<char> has;     // has[k] = 1 nếu tag[k] khác ánh xạ đồng nhất

void compose(int k, const Map &t) {  // áp t sau tag[k]
    for (int x = 0; x < 26; x++) tag[k][x] = t[tag[k][x]];
    has[k] = 1;
}

void push(int k) {
    if (!has[k]) return;
    // thao tác ở nút cha mới hơn thao tác đang chờ ở nút con -> áp sau
    compose(2 * k, tag[k]);
    compose(2 * k + 1, tag[k]);
    for (int x = 0; x < 26; x++) tag[k][x] = x;
    has[k] = 0;
}

void apply_block(int blk) {  // áp tag của lá vào các ký tự của khối
    int k = blk + sz;
    if (!has[k]) return;
    int a = blk * B, b = min(a + B, n);
    for (int p = a; p < b; p++) s[p] = 'a' + tag[k][s[p] - 'a'];
    for (int x = 0; x < 26; x++) tag[k][x] = x;
    has[k] = 0;
}

void materialize(int blk) {  // đẩy hết thao tác đang chờ xuống khối blk
    int k = blk + sz;
    for (int h = lg; h > 0; h--) push(k >> h);
    apply_block(blk);
}

void change(int l, int r, char c1, char c2) {  // đổi c1 -> c2 trên s[l..r]
    for (int p = l; p <= r; p++)
        if (s[p] == c1) s[p] = c2;
}

string solve_biendoixau(const vector<int> &qi, const vector<int> &qj,
                        const vector<char> &qc1, const vector<char> &qc2) {
    n = s.size();
    int nb = (n + B - 1) / B;
    sz = 1, lg = 0;
    while (sz < nb) sz *= 2, lg++;
    Map id;
    for (int x = 0; x < 26; x++) id[x] = x;
    tag.assign(2 * sz, id);
    has.assign(2 * sz, 0);
    for (size_t q = 0; q < qi.size(); q++) {
        int i = qi[q], j = qj[q];
        char c1 = qc1[q], c2 = qc2[q];
        int bi = i / B, bj = j / B;
        if (bi == bj) {
            materialize(bi);
            change(i, j, c1, c2);
            continue;
        }
        int lo = bi, hi = bj;
        if (i != bi * B) {  // khối trái chỉ phủ một phần
            materialize(bi);
            change(i, (bi + 1) * B - 1, c1, c2);
            lo = bi + 1;
        }
        if (j + 1 != min((bj + 1) * B, n)) {  // khối phải chỉ phủ một phần
            materialize(bj);
            change(bj * B, j, c1, c2);
            hi = bj - 1;
        }
        if (lo > hi) continue;
        Map g = id;
        g[c1 - 'a'] = c2 - 'a';
        // gán thao tác cho các khối nguyên [lo, hi] (cây phân đoạn lười, kiểu từ dưới lên)
        int l = lo + sz, r = hi + 1 + sz;
        for (int h = lg; h > 0; h--) {
            if (((l >> h) << h) != l) push(l >> h);
            if (((r >> h) << h) != r) push((r - 1) >> h);
        }
        while (l < r) {
            if (l & 1) compose(l++, g);
            if (r & 1) compose(--r, g);
            l >>= 1;
            r >>= 1;
        }
    }
    // đẩy mọi thao tác xuống lá rồi áp vào từng khối
    for (int k = 1; k < sz; k++) push(k);
    for (int blk = 0; blk < nb; blk++) apply_block(blk);
    return s;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    // Có input.inp thì đọc/ghi file, không thì dùng bàn phím/màn hình
    if (FILE *f = fopen("input.inp", "r")) {
        fclose(f);
        freopen("input.inp", "r", stdin);
        freopen("output.out", "w", stdout);
    }
    int m;
    cin >> s >> m;
    vector<int> qi(m), qj(m);
    vector<char> qc1(m), qc2(m);
    for (int q = 0; q < m; q++) cin >> qi[q] >> qj[q] >> qc1[q] >> qc2[q];
    cout << solve_biendoixau(qi, qj, qc1, qc2) << "\n";
    return 0;
}

    2. Ma trận¶

    Ý tưởng (Freivalds). Tính \(A^3\) trực tiếp tốn \(O(n^3)\), với \(n = 500\) và \(T = 20\) là \(2 \cdot 20 \cdot 500^3 = 5 \cdot 10^9\) phép nhân, quá chậm. Thay vào đó chọn ngẫu nhiên vectơ \(v\) và so sánh

    \[A \cdot (A \cdot (A \cdot v)) \quad \text{với} \quad B \cdot v.\]

    Mỗi phép nhân ma trận với vectơ chỉ tốn \(O(n^2)\), tổng cộng 4 lần. Nếu \(A^3 = B\) thì hai vế luôn bằng nhau. Nếu \(A^3 \ne B\), gọi \(D = A^3 - B \ne 0\), hai vế chỉ bằng nhau khi \(Dv = 0\), xác suất rất nhỏ khi \(v\) ngẫu nhiên.

    Phần tử của \(Bv\) có thể tới \(500 \cdot 10^{18} \cdot |v_i|\), tràn long long, nên mọi phép tính lấy dư theo số nguyên tố \(P = 2^{61} - 1 \approx 2.3 \cdot 10^{18}\) và chọn \(v_i\) ngẫu nhiên trong \([0, P)\):

    • Phần tử của \(A^3\) có trị tuyệt đối không quá \(n^2 \cdot 1000^3 = 2.5 \cdot 10^{14}\), của \(B\) không quá \(10^{18}\), nên \(|d_{i,j}| < P\). Phần tử \(d_{i,j} \ne 0\) thì vẫn khác 0 khi lấy dư \(P\).
    • Trên trường số dư theo \(P\), với \(D \ne 0\) và \(v\) ngẫu nhiên đều, xác suất \(Dv \equiv 0\) không quá \(1/P\). Lời giải thử 2 lần cho mỗi bộ dữ liệu.

    Bẫy thường gặp

    • Nhân ma trận \(O(n^3)\) chỉ qua subtask \(n \le 10\).
    • Freivalds với \(v\) gồm 0/1 và không lấy dư: \(Bv\) tràn long long vì \(|b_{i,j}|\) tới \(10^{18}\).
    • Số âm khi lấy dư trong C++: x % P có thể âm, phải cộng thêm \(P\).
    • Đọc dữ liệu: test lớn nhất có \(20 \cdot 2 \cdot 500^2 = 10^7\) số, hơn 100 MB. cin thường không kịp 0,5 giây; bản C++ đọc bằng fread.

    Nếu viết bằng C++

    Nhân hai số dư theo \(2^{61} - 1\) cần unsigned __int128 (có trên GCC/Clang). Trên máy kiểm thử, bản C++ chạy test 20 bộ \(n = 500\) khoảng 0,15 giây; bản Python mất 3-7 giây (riêng việc đọc \(10^7\) số đã quá 0,5 giây), nên Python chỉ hợp với subtask \(n \le 10\).

    matran.py
    import random
import sys

P = (1 << 61) - 1  # số nguyên tố Mersenne, lớn hơn mọi |A^3 - B| có thể xảy ra
ROUNDS = 2         # số lần thử Freivalds cho mỗi bộ dữ liệu


def mat_vec(m, v):
    # Nhân ma trận m (danh sách dòng) với vectơ v, lấy dư P
    return [sum(map(int.__mul__, row, v)) % P for row in m]


def solve_matran(data):
    tokens = data.split()
    t = int(tokens[0])
    pos = 1
    out = []
    for _ in range(t):
        n = int(tokens[pos])
        pos += 1
        a = []
        for _ in range(n):
            a.append(list(map(int, tokens[pos:pos + n])))
            pos += n
        b = []
        for _ in range(n):
            b.append([x % P for x in map(int, tokens[pos:pos + n])])
            pos += n
        ok = True
        for _ in range(ROUNDS):
            # Freivalds: so sánh A(A(Av)) với Bv cho vectơ ngẫu nhiên v
            v = [random.randrange(P) for _ in range(n)]
            if mat_vec(a, mat_vec(a, mat_vec(a, v))) != mat_vec(b, v):
                ok = False
                break
        out.append("YES" if ok else "NO")
    return "\n".join(out)


if __name__ == "__main__":
    try:
        with open("input.inp") as f:
            data = f.read()
        with open("output.out", "w") as f:
            f.write(solve_matran(data) + "\n")
    except FileNotFoundError:
        print(solve_matran(sys.stdin.read()))
    matran.cpp
    #include <chrono>
#include <cstdio>
#include <random>
#include <string>
#include <vector>
using namespace std;

typedef unsigned long long ull;
const ull P = (1ULL << 61) - 1;  // số nguyên tố Mersenne, lớn hơn mọi |A^3 - B| có thể xảy ra
const int ROUNDS = 2;            // số lần thử Freivalds cho mỗi bộ dữ liệu

ull mulmod(ull a, ull b) {
    unsigned __int128 x = (unsigned __int128)a * b;
    ull r = (ull)(x & P) + (ull)(x >> 61);
    return r >= P ? r - P : r;
}

ull to_mod(long long x) {  // đưa số âm về [0, P)
    long long r = x % (long long)P;
    return (ull)(r < 0 ? r + (long long)P : r);
}

// Nhân ma trận m (n x n, lưu phẳng) với vectơ v, lấy dư P
vector<ull> mat_vec(const vector<ull> &m, const vector<ull> &v, int n) {
    vector<ull> r(n);
    for (int i = 0; i < n; i++) {
        ull s = 0;
        const ull *row = &m[(size_t)i * n];
        for (int j = 0; j < n; j++) {
            s += mulmod(row[j], v[j]);
            if (s >= P) s -= P;
        }
        r[i] = s;
    }
    return r;
}

        buf_len = fread(buf, 1, sizeof(buf), stdin);
        buf_pos = 0;
        if (buf_len == 0) return -1;
    }

}

long long read_int() {
    int c = read_char();
    while (c != '-' && (c < '0' || c > '9')) c = read_char();
    bool neg = false;
    if (c == '-') {
        neg = true;
        c = read_char();
    }
    long long x = 0;
    while (c >= '0' && c <= '9') {
        x = x * 10 + (c - '0');
        c = read_char();
    }
    return neg ? -x : x;
}

    3. Cắt ghép xâu¶

    Nhận xét. Có cách cắt ghép khi và chỉ khi \(S_1\) và \(S_2\) có cùng tập ký tự (cùng độ dài, mỗi chữ cái xuất hiện cùng số lần): khi đó cắt \(S_1\) thành từng ký tự là ghép được. Ngược lại in -1.

    Tìm số đoạn ít nhất là bài toán phân hoạch xâu chung nhỏ nhất (Minimum Common String Partition), được biết là NP-khó, nên đề chấm theo tỉ lệ \(\min\{1, q^5/c^5\}\) so với kết quả của Ban giám khảo. Lời giải tham khảo dùng tham lam, không bảo đảm tối ưu.

    Ý tưởng tham lam. Lặp lại: tìm độ dài \(L\) lớn nhất mà phần chưa dùng của \(S_1\) và \(S_2\) còn chung một đoạn độ dài \(L\), rồi ghép cặp các đoạn đó và đánh dấu đã dùng.

    • Kiểm tra "có đoạn chung độ dài \(L\)" bằng băm xâu (hash đa thức mô-đun \(2^{61} - 1\)): đưa giá trị băm của mọi cửa sổ độ dài \(L\) chưa dùng của \(S_1\) vào tập, rồi tra các cửa sổ của \(S_2\). Mỗi lần kiểm tra \(O(n)\).
    • \(L\) chỉ giảm dần qua các lượt nên tìm \(L\) bằng cách dò lùi với bước nhân đôi rồi chặt nhị phân.
    • Với \(L\) đã chọn, duyệt các cửa sổ của \(S_2\), mỗi cửa sổ còn trống thì ghép với một vị trí còn trống ở \(S_1\) có cùng giá trị băm (so sánh trực tiếp để tránh trùng băm). Sau một lượt như vậy không còn cặp đoạn chung độ dài \(L\) nào, sang \(L\) nhỏ hơn.
    • Bỏ hai đoạn giống nhau khỏi \(S_1\) và \(S_2\) thì phần còn lại vẫn cùng tập ký tự, nên đến \(L = 1\) luôn ghép hết.
    • Cuối cùng gộp hai đoạn liền nhau trong \(S_1\) mà cũng liền nhau, đúng thứ tự trong \(S_2\) (giảm thêm điểm cắt).

    Kết quả tham lam phụ thuộc cách chọn khi có nhiều đoạn cùng độ dài, nên chạy 4 biến thể (duyệt \(S_2\) từ trái hoặc từ phải; ưu tiên vị trí đầu tiên hoặc cuối cùng ở \(S_1\)) và lấy phương án ít đoạn nhất.

    Với ví dụ abcabc / aabbcc: đoạn chung dài nhất có độ dài 2 (ab, bc), ghép xong hai đoạn đó còn lại a, c; tổng 4 đoạn, \(c = 3\) bằng ví dụ của đề (cách cắt có thể khác).

    Bẫy thường gặp

    • Quên trường hợp -1: hai xâu khác độ dài hoặc khác số lần xuất hiện của một chữ cái.
    • In vị trí cắt sai quy ước: \(p\) là "cắt sau ký tự thứ \(p\)", đánh số từ 1. Đoạn bắt đầu ở chỉ số \(a\) (đánh số từ 0) ứng với vị trí cắt \(p = a\).
    • Dòng thứ ba là thứ tự ghép các đoạn (đoạn nào đứng đầu \(S_2\), rồi đoạn nào...), không phải vị trí của mỗi đoạn trong \(S_2\).
    • Băm trùng giá trị: luôn so sánh thật hai đoạn trước khi ghép, nếu không có thể in phương án sai.

    Kết quả kiểm chứng

    Trên 345 test ngẫu nhiên \(|S_1| \le 9\) so với vét cạn mọi cách cắt, lời giải luôn in phương án hợp lệ, đạt tối ưu ở 337 test (8 test thừa 1 điểm cắt), điểm trung bình theo công thức của đề khoảng 98%.     import sys

M = (1 << 61) - 1   # mô-đun băm
BASE = 131


def prefix_hash(s):
    h = [0] * (len(s) + 1)
    for i, ch in enumerate(s):
        h[i + 1] = (h[i] * BASE + ord(ch)) % M
    return h


def greedy(s1, s2, rev, last):
    # Tham lam: lặp lại việc ghép đoạn chung DÀI NHẤT còn lại của S1 và S2.
    # rev: duyệt S2 từ phải sang; last: ưu tiên vị trí xuất hiện sau cùng ở S1.
    n = len(s1)
    h1, h2 = prefix_hash(s1), prefix_hash(s2)
    pw = [1] * (n + 1)
    for i in range(n):
        pw[i + 1] = pw[i] * BASE % M
    used1, used2 = bytearray(n), bytearray(n)
    pieces = []          # (vị trí trong S1, vị trí trong S2, độ dài)
    remain = n

    def windows(h, pu, L):
        # (băm, vị trí) của các cửa sổ độ dài L chưa dùng ký tự nào
        p = pw[L]
        return [((h[i + L] - h[i] * p) % M, i)
                for i in range(n - L + 1) if pu[i + L] == pu[i]]

    hi = n
    while remain > 0:
        pu1, pu2 = [0] * (n + 1), [0] * (n + 1)
        for i in range(n):
            pu1[i + 1] = pu1[i] + used1[i]
            pu2[i + 1] = pu2[i] + used2[i]

        def exists(L):
            st = {x for x, _ in windows(h1, pu1, L)}
            return any(x in st for x, _ in windows(h2, pu2, L))

        # tìm độ dài L lớn nhất còn đoạn chung: L chỉ giảm dần nên dò lùi
        # từ hi với bước nhân đôi, rồi chặt nhị phân trong khoảng tìm được
        L = hi
        if not exists(hi):
            bad, step = hi, 1
            while True:
                c = max(1, hi - step)
                if c == 1 or exists(c):  # độ dài 1 luôn có đoạn chung
                    break
                bad, step = c, step * 2
            lo, up = c, bad - 1
            while lo < up:
                mid = (lo + up + 1) // 2
                if exists(mid):
                    lo = mid
                else:
                    up = mid - 1
            L = lo
        # lấy mọi cặp đoạn chung độ dài L không chồng nhau
        cand = {}
        for x, i in windows(h1, pu1, L):
            cand.setdefault(x, []).append(i)
        if last:
            for lst in cand.values():
                lst.reverse()
        w2 = windows(h2, pu2, L)
        if rev:
            w2.reverse()
        for x, p in w2:
            if used2[p] or used2[p + L - 1]:
                continue  # chồng lên đoạn vừa lấy trong lượt này
            lst = cand.get(x)
            if not lst:
                continue
            k = 0
            while k < len(lst):
                a = lst[k]
                if used1[a] or used1[a + L - 1]:
                    del lst[k]  # đoạn đã bị dùng: bỏ khỏi danh sách
                    continue
                if s1[a:a + L] == s2[p:p + L]:  # kiểm tra thật, tránh trùng băm
                    used1[a:a + L] = b"\1" * L
                    used2[p:p + L] = b"\1" * L
                    pieces.append((a, p, L))
                    remain -= L
                    del lst[k]
                    break
                k += 1
        hi = max(1, L - 1)
    # gộp hai đoạn kề nhau ở S1 mà cũng kề nhau, đúng thứ tự ở S2
    pieces.sort()
    merged = [list(pieces[0])]
    for a, p, L in pieces[1:]:
        last_piece = merged[-1]
        if last_piece[1] + last_piece[2] == p:
            last_piece[2] += L
        else:
            merged.append([a, p, L])
    return merged


def solve_catghepxau(data):
    tokens = data.split()
    s1, s2 = tokens[0], tokens[1]
    if len(s1) != len(s2) or sorted(s1) != sorted(s2):
        return "-1"  # khác tập ký tự thì không thể cắt ghép
    best = None
    for rev in (0, 1):          # chạy 4 biến thể, giữ phương án ít đoạn nhất
        for last in (0, 1):
            cur = greedy(s1, s2, rev, last)
            if best is None or len(cur) < len(best):
                best = cur
    cuts = [a for a, _, _ in best[1:]]   # cắt sau ký tự thứ a (đánh số từ 1)
    order = sorted(range(len(best)), key=lambda k: best[k][1])
    return "\n".join([str(len(cuts)),
                      " ".join(map(str, cuts)),
                      " ".join(str(k + 1) for k in order)])

#include <array>
#include <cstdio>
#include <iostream>
#include <string>
#include <unordered_map>
#include <unordered_set>
#include <vector>
using namespace std;

typedef unsigned long long ull;
typedef array<int, 3> Piece;  // (vị trí trong S1, vị trí trong S2, độ dài)
const ull M = (1ULL << 61) - 1;  // mô-đun băm
const ull BASE = 131;

ull mulmod(ull a, ull b) {
    unsigned __int128 x = (unsigned __int128)a * b;
    ull r = (ull)(x & M) + (ull)(x >> 61);
    return r >= M ? r - M : r;
}

vector<ull> prefix_hash(const string &s) {
    vector<ull> h(s.size() + 1, 0);
    for (size_t i = 0; i < s.size(); i++) {
        h[i + 1] = mulmod(h[i], BASE) + (unsigned char)s[i];
        if (h[i + 1] >= M) h[i + 1] -= M;
    }
    return h;
}

// Tham lam: lặp lại việc ghép đoạn chung DÀI NHẤT còn lại của S1 và S2.
// rev: duyệt S2 từ phải sang; last: ưu tiên vị trí xuất hiện sau cùng ở S1.
vector<Piece> greedy(const string &s1, const string &s2, bool rev, bool last) {
    int n = s1.size();
    vector<ull> h1 = prefix_hash(s1), h2 = prefix_hash(s2), pw(n + 1, 1);
    for (int i = 0; i < n; i++) pw[i + 1] = mulmod(pw[i], BASE);
    vector<char> used1(n, 0), used2(n, 0);
    vector<int> pu1(n + 1, 0), pu2(n + 1, 0);
    vector<Piece> pieces;
    int remain = n, hi = n;

    // (băm, vị trí) của các cửa sổ độ dài L chưa dùng ký tự nào
    auto windows = [&](const vector<ull> &h, const vector<int> &pu, int L) {
        vector<pair<ull, int>> w;
        for (int i = 0; i + L <= n; i++)
            if (pu[i + L] == pu[i]) w.push_back({(h[i + L] + M - mulmod(h[i], pw[L])) % M, i});
        return w;
    };
    while (remain > 0) {
        for (int i = 0; i < n; i++) {
            pu1[i + 1] = pu1[i] + used1[i];
            pu2[i + 1] = pu2[i] + used2[i];
        }
        auto exists = [&](int L) {
            unordered_set<ull> st;
            for (auto &e : windows(h1, pu1, L)) st.insert(e.first);
            for (auto &e : windows(h2, pu2, L))
                if (st.count(e.first)) return true;
            return false;
        };
        // tìm độ dài L lớn nhất còn đoạn chung: L chỉ giảm dần nên dò lùi
        // từ hi với bước nhân đôi, rồi chặt nhị phân trong khoảng tìm được
        int L = hi;
        if (!exists(hi)) {
            int bad = hi, step = 1, c;
            while (true) {
                c = max(1, hi - step);
                if (c == 1 || exists(c)) break;  // độ dài 1 luôn có đoạn chung
                bad = c;
                step *= 2;
            }
            int lo = c, up = bad - 1;
            while (lo < up) {
                int mid = (lo + up + 1) / 2;
                if (exists(mid)) lo = mid;
                else up = mid - 1;
            }
            L = lo;
        }
        // lấy mọi cặp đoạn chung độ dài L không chồng nhau
        unordered_map<ull, vector<int>> cand;
        for (auto &e : windows(h1, pu1, L)) cand[e.first].push_back(e.second);
        if (last)
            for (auto &kv : cand) reverse(kv.second.begin(), kv.second.end());
        vector<pair<ull, int>> w2 = windows(h2, pu2, L);
        if (rev) reverse(w2.begin(), w2.end());
        for (auto &e : w2) {
            int p = e.second;
            if (used2[p] || used2[p + L - 1]) continue;  // chồng lên đoạn vừa lấy trong lượt này
            auto it = cand.find(e.first);
            if (it == cand.end()) continue;
            vector<int> &lst = it->second;
            size_t k = 0;
            while (k < lst.size()) {
                int a = lst[k];
                if (used1[a] || used1[a + L - 1]) {  // đoạn đã bị dùng: bỏ khỏi danh sách
                    lst.erase(lst.begin() + k);
                    continue;
                }
                if (s1.compare(a, L, s2, p, L) == 0) {  // kiểm tra thật, tránh trùng băm
                    fill(used1.begin() + a, used1.begin() + a + L, 1);
                    fill(used2.begin() + p, used2.begin() + p + L, 1);
                    pieces.push_back({a, p, L});
                    remain -= L;
                    lst.erase(lst.begin() + k);
                    break;
                }
                k++;
            }
        }
        hi = max(1, L - 1);
    }
    // gộp hai đoạn kề nhau ở S1 mà cũng kề nhau, đúng thứ tự ở S2
    sort(pieces.begin(), pieces.end());
    vector<Piece> merged = {pieces[0]};
    for (size_t k = 1; k < pieces.size(); k++) {
        Piece &lp = merged.back();
        if (lp[1] + lp[2] == pieces[k][1]) lp[2] += pieces[k][2];
        else merged.push_back(pieces[k]);
    }
    return merged;
}

string solve_catghepxau(const string &s1, const string &s2) {
    string t1 = s1, t2 = s2;
    sort(t1.begin(), t1.end());
    sort(t2.begin(), t2.end());
    if (t1 != t2) return "-1";  // khác tập ký tự thì không thể cắt ghép
    vector<Piece> best;
    for (int rev = 0; rev < 2; rev++)  // chạy 4 biến thể, giữ phương án ít đoạn nhất
        for (int last = 0; last < 2; last++) {
            vector<Piece> cur = greedy(s1, s2, rev, last);
            if (best.empty() || cur.size() < best.size()) best = cur;
        }
    int c = best.size() - 1;
    string out = to_string(c) + "\n";
    for (int k = 1; k <= c; k++) {  // cắt sau ký tự thứ a (đánh số từ 1)
        if (k > 1) out += ' ';
        out += to_string(best[k][0]);
    }
    vector<int> order(c + 1);
    for (int k = 0; k <= c; k++) order[k] = k;
    sort(order.begin(), order.end(), [&](int x, int y) { return best[x][1] < best[y][1]; });
    out += "\n";
    for (int k = 0; k <= c; k++) {
        if (k) out += ' ';
        out += to_string(order[k] + 1);
    }
    return out;
}

    4. Dãy cấp số nhân¶

    Nhận xét. Vì \(q \ge 2\) và \(1 \le a_i \le 10^9 < 2^{30}\), một dãy cấp số nhân công bội \(q\) trong dãy có tối đa \(30\) phần tử (dài nhất là \(1, 2, 4, \ldots, 2^{29}\)). Lời giải dùng mảng độ dài 32 cho an toàn; mọi \(k\) lớn hơn có đáp án 0.

    Ý tưởng (quy hoạch động). Duyệt \(a_1, a_2, \ldots\) theo thứ tự. Gọi \(cnt[v][L]\) là số dãy con cấp số nhân độ dài \(L\) kết thúc bằng giá trị \(v\) trong phần đã duyệt. Khi gặp \(a_i = v\):

    • \(v\) tự nó là dãy độ dài 1;
    • nếu \(q \mid v\), mọi dãy độ dài \(L - 1\) kết thúc bằng \(v / q\) nối thêm \(v\) thành dãy độ dài \(L\): thêm \(cnt[v/q][L-1]\) dãy.

    Cộng các số mới vào \(cnt[v][\cdot]\) và vào tổng \(total[L]\). Lưu \(cnt\) trong từ điển (dict / unordered_map) theo giá trị vì \(a_i\) tới \(10^9\).

    Với ví dụ 1 2 8 4 2, \(q = 2\): gặp \(2\) (vị trí 2) nối sau \(1\) được 1 dãy độ dài 2; gặp \(4\) nối sau các dãy kết thúc bằng 2 được \((2, 4)\) và \((1, 2, 4)\); gặp \(2\) (vị trí 5) nối sau \(1\) được thêm \((1, 2)\). Tổng: độ dài 2 có 3 dãy, độ dài 3 có 1 dãy.

    Bẫy thường gặp

    • Nối sai chiều: chỉ được nối \(v\) vào sau dãy kết thúc bằng \(v/q\) đã xuất hiện trước \(v\). Duyệt theo thứ tự chỉ số và cập nhật \(cnt[v]\) sau khi tính xong là đúng.
    • Tính \(v \cdot q\) thay vì \(v / q\) có thể tràn kiểu 32 bit (\(10^9 \cdot 10^9\)).
    • Phải in đủ \(n - 1\) số, kể cả các số 0 với \(k > 30\).
    • Hai phần tử bằng nhau ở hai vị trí khác nhau tạo hai dãy khác nhau (ví dụ có hai dãy \((1, 2)\)).
    daycapsonhan.py
    import sys

MOD = 10**9 + 7
K = 32  # q >= 2, a_i <= 10^9 nên dãy cấp số nhân dài tối đa 30 phần tử


def solve_daycapsonhan(data):
    tokens = data.split()
    n, q = int(tokens[0]), int(tokens[1])
    a = tokens[2:2 + n]
    # cnt[v][L] = số dãy cấp số nhân độ dài L kết thúc bằng giá trị v (đã duyệt)
    cnt = {}
    total = [0] * (K + 1)
    for t in a:
        v = int(t)
        cur = [0] * (K + 1)
        cur[1] = 1
        if v % q == 0:
            prev = cnt.get(v // q)
            if prev is not None:
                # nối v vào sau các dãy kết thúc bằng v / q
                for L in range(2, K + 1):
                    cur[L] = prev[L - 1]
        own = cnt.get(v)
        if own is None:
            own = [0] * (K + 1)
            cnt[v] = own
        for L in range(1, K + 1):
            if cur[L]:
                own[L] = (own[L] + cur[L]) % MOD
                total[L] = (total[L] + cur[L]) % MOD
    res = []
    for k in range(2, n + 1):
        res.append(total[k] if k <= K else 0)
    return " ".join(map(str, res))

    try:
        with open("input.inp") as f:
            data = f.read()

            f.write(solve_daycapsonhan(data) + "\n")
    except FileNotFoundError:
        print(solve_daycapsonhan(sys.stdin.read()))
    daycapsonhan.cpp
    #include <array>
#include <cstdio>
#include <iostream>
#include <unordered_map>
#include <vector>
using namespace std;

const long long MOD = 1000000007LL;
const int K = 32;  // q >= 2, a_i <= 10^9 nên dãy cấp số nhân dài tối đa 30 phần tử

vector<long long> solve_daycapsonhan(int n, long long q, const vector<long long> &a) {
    // cnt[v][L] = số dãy cấp số nhân độ dài L kết thúc bằng giá trị v (đã duyệt)
    unordered_map<long long, array<long long, K + 1>> cnt;
    cnt.reserve(2 * n + 1);
    array<long long, K + 1> total{};
    for (long long v : a) {
        array<long long, K + 1> cur{};
        cur[1] = 1;
        if (v % q == 0) {
            auto it = cnt.find(v / q);
            if (it != cnt.end())  // nối v vào sau các dãy kết thúc bằng v / q
                for (int L = 2; L <= K; L++) cur[L] = it->second[L - 1];
        }
        auto &own = cnt[v];  // phần tử mới được khởi tạo bằng 0
        for (int L = 1; L <= K; L++) {
            if (cur[L]) {
                own[L] = (own[L] + cur[L]) % MOD;
                total[L] = (total[L] + cur[L]) % MOD;
            }
        }
    }
    vector<long long> res;
    for (int k = 2; k <= n; k++) res.push_back(k <= K ? total[k] : 0);
    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    // Có input.inp thì đọc/ghi file, không thì dùng bàn phím/màn hình
    if (FILE *f = fopen("input.inp", "r")) {
        fclose(f);
        freopen("input.inp", "r", stdin);
        freopen("output.out", "w", stdout);
    }
    int n;
    long long q;
    cin >> n >> q;
    vector<long long> a(n);
    for (auto &x : a) cin >> x;
    vector<long long> res = solve_daycapsonhan(n, q, a);
    for (size_t i = 0; i < res.size(); i++) {
        if (i) cout << ' ';
        cout << res[i];
    }
    cout << "\n";
    return 0;
}
