    Đáp án - Bảng C Chung kết toàn quốc 2021¶

    Lời giải tham khảo bằng Python 3 và C++ (C++17). Xem lại đề bài.

    Đề chính thức đọc/ghi chuẩn. Để đồng bộ với các đề khác trên site, mỗi chương trình đọc input.inp và ghi output.out nếu có file input.inp; nếu không có thì đọc từ bàn phím và in ra màn hình, đúng như đề yêu cầu.

    STT Bài Ý tưởng Độ phức tạp
    1 Bài tập Sắp xếp; \(m < n\): cửa sổ trượt lấy max; \(m > n\): chặt nhị phân \(d\) \(O(n \log n + n \log C)\)
    2 Bài khó Tìm kiếm sâu dần theo \(k\), hai số cuối giải bằng duyệt ước tìm kiếm có cắt nhánh
    3 Xâu con Đếm cặp cách chọn cho cùng một xâu con, quy hoạch động 2 chiều + tổng tiền tố \(O(N^2)\)
    4 Cắt bánh Thêm từng đường cắt, số miếng tăng \(1 +\) số giao điểm phân biệt mới \(O(m + D^2 \log D)\), \(D \le 1225\)

    1. Bài tập¶

    Ý tưởng. Sắp xếp độ khó tăng dần, gọi \(g_i = c_{i+1} - c_i\) là chênh lệch giữa hai bài liền kề.

    Trường hợp \(m < n\) (bỏ bài). Giữ lại \(m\) bài liên tiếp trong dãy đã sắp xếp luôn tốt nhất. Giả sử giữ các bài có chỉ số \(i_1 < i_2 < \ldots < i_m\). Xét \(m\) bài liên tiếp \(i_1, i_1 + 1, \ldots, i_1 + m - 1\): mọi chênh lệch giữa chúng là một \(g_j\) nằm trong đoạn \([i_1, i_m]\), và \(g_j\) đó không vượt quá chênh lệch của cặp bài giữ lại "bao" nó. Vậy đáp án là

    \[\min_{\text{cửa sổ } m - 1 \text{ khoảng liên tiếp}} \ \max g_j,\]

    tính bằng max trên cửa sổ trượt (deque) trong \(O(n)\).

    Trường hợp \(m > n\) (thêm bài). Muốn mọi chênh lệch \(\le d\) thì khoảng \(g\) cần chen thêm \(\lceil g / d \rceil - 1\) bài. Đặt thêm bài ra ngoài khoảng \([c_{\min}, c_{\max}]\) chỉ tạo thêm chênh lệch, còn bài thừa có thể đặt trùng độ khó sẵn có (chênh lệch 0). Nên \(d\) khả thi khi \(\sum \lceil g_i / d \rceil - 1 \le m - n\). Điều kiện này đơn điệu theo \(d\), nên chặt nhị phân \(d\) trên \([1, \max g]\). Nếu mọi bài cùng độ khó thì đáp án là 0.

    Ví dụ 2: sắp xếp được \(6, 8, 9\); thêm một bài độ khó 7 thì dãy \(6, 7, 8, 9\) có \(d = 1\).

    Bẫy thường gặp

    • Khi \(m < n\), chọn tham lam "bỏ bài gây chênh lệch lớn nhất" là sai; bài bỏ ở giữa còn làm gộp hai khoảng lại.
    • Với \(g = 0\) (hai bài trùng độ khó), công thức \(\lfloor (g - 1) / d \rfloor\) cho \(-1\) trong Python. Lời giải chỉ cộng khi \(g > d\).
    • Đáp án có thể là 0 (ví dụ có hai bài trùng độ khó và \(m < n\)).

    Nếu viết bằng C++

    \(c_i \le 10^9\) nên chênh lệch vừa int, nhưng tổng số bài cần chen thêm có thể lớn; lời giải dùng long long cho chắc.

    tasks.py
    import sys
from collections import deque


def solve_tasks(data):
    tokens = data.split()
    n, m = int(tokens[0]), int(tokens[1])
    c = sorted(int(x) for x in tokens[2:2 + n])
    g = [c[i + 1] - c[i] for i in range(n - 1)]  # chênh lệch hai bài liền kề

    if m < n:
        # Giữ m bài liên tiếp (sau sắp xếp) là tối ưu -> cửa sổ m-1 khoảng liền kề.
        # Lấy min của max trên mọi cửa sổ bằng deque (max trượt).
        w = m - 1
        dq = deque()
        best = None
        for i in range(n - 1):
            while dq and g[dq[-1]] <= g[i]:
                dq.pop()
            dq.append(i)
            if dq[0] <= i - w:
                dq.popleft()
            if i >= w - 1:
                if best is None or g[dq[0]] < best:
                    best = g[dq[0]]
        return str(best)

    # m > n: thêm k bài. Khoảng g cần ceil(g/d) - 1 bài chen vào để mọi khoảng <= d.
    # Bài thừa đặt trùng độ khó đã có (chênh lệch 0) nên không ảnh hưởng.
    k = m - n
    mx = max(g)
    if mx == 0:
        return "0"
    lo, hi = 1, mx
    while lo < hi:
        d = (lo + hi) // 2
        need = 0
        for x in g:
            if x > d:
                need += (x - 1) // d
        if need <= k:
            hi = d
        else:
            lo = d + 1
    return str(lo)


if __name__ == "__main__":
    try:
        with open("input.inp") as f:
            data = f.read()
        with open("output.out", "w") as f:
            f.write(solve_tasks(data) + "\n")
    except FileNotFoundError:
        print(solve_tasks(sys.stdin.read()))
    tasks.cpp
    #include <algorithm>
#include <cstdio>
#include <deque>
#include <iostream>
#include <vector>
using namespace std;

long long solve_tasks(int n, int m, vector<long long> c) {
    sort(c.begin(), c.end());
    vector<long long> g(n - 1);  // chênh lệch hai bài liền kề
    for (int i = 0; i + 1 < n; i++) g[i] = c[i + 1] - c[i];

    if (m < n) {
        // Giữ m bài liên tiếp (sau sắp xếp) là tối ưu -> cửa sổ m-1 khoảng liền kề.
        // Lấy min của max trên mọi cửa sổ bằng deque (max trượt).
        int w = m - 1;
        deque<int> dq;
        long long best = -1;
        for (int i = 0; i < n - 1; i++) {
            while (!dq.empty() && g[dq.back()] <= g[i]) dq.pop_back();
            dq.push_back(i);
            if (dq.front() <= i - w) dq.pop_front();
            if (i >= w - 1 && (best < 0 || g[dq.front()] < best)) best = g[dq.front()];
        }
        return best;
    }

    // m > n: thêm k bài. Khoảng g cần ceil(g/d) - 1 bài chen vào để mọi khoảng <= d.
    // Bài thừa đặt trùng độ khó đã có (chênh lệch 0) nên không ảnh hưởng.
    long long k = m - n;
    long long mx = *max_element(g.begin(), g.end());
    if (mx == 0) return 0;
    long long lo = 1, hi = mx;
    while (lo < hi) {
        long long d = (lo + hi) / 2;
        long long need = 0;
        for (long long x : g)
            if (x > d) need += (x - 1) / d;
        if (need <= k) hi = d;
        else lo = d + 1;
    }
    return lo;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    // Có input.inp thì đọc/ghi file, không thì dùng bàn phím/màn hình
    if (FILE *f = fopen("input.inp", "r")) {
        fclose(f);
        freopen("input.inp", "r", stdin);
        freopen("output.out", "w", stdout);
    }
    int n, m;
    cin >> n >> m;
    vector<long long> c(n);
    for (auto &x : c) cin >> x;
    cout << solve_tasks(n, m, c) << "\n";
    return 0;
}

    2. Bài khó¶

    Ý tưởng. Chia hai vế cho \(a_1 a_2 \cdots a_k\):

    \[\prod_{i=1}^{k} \frac{a_i + t}{a_i} = n, \qquad \frac{a_i + t}{a_i} = 1 + \frac{t}{a_i} \in (1,\ t + 1].\]

    Luôn có nghiệm khi \(n \ge 2\). Chọn \(a = j \cdot t\) thì \(\frac{a + t}{a} = \frac{j + 1}{j}\), và \(\frac{2}{1} \cdot \frac{3}{2} \cdots \frac{n}{n - 1} = n\). Vậy chỉ có \(n = 1\) là in \(-1\) (mỗi thừa số đều lớn hơn 1).

    Tìm \(k\) nhỏ nhất. Thử lần lượt \(k = 1, 2, 3, \ldots\) (tìm kiếm sâu dần), \(k\) đầu tiên tìm được là đáp án. Với mỗi \(k\), kiểm tra phân số \(r = p/q\) (tối giản, ban đầu \(r = n\)) có viết được thành tích \(k\) thừa số dạng \(\frac{a + t}{a}\) hay không, với \(a_1 \le a_2 \le \ldots \le a_k\) (thừa số giảm dần):

    • Còn 1 thừa số: \(\frac{a + t}{a} = \frac{p}{q} \iff a = \frac{t q}{p - q}\), kiểm tra chia hết và \(a \ge a_{\text{trước}}\).
    • Còn 2 thừa số \(a \le b\): biến đổi \(p a b = q (a + t)(b + t)\) thành $\((a u - q t)(b u - q t) = p \, q \, t^2, \qquad u = p - q.\)$ Duyệt các ước \(X \le \sqrt{p q t^2}\), thử \(X \equiv -q t \pmod u\) để lấy \(a = \frac{X + q t}{u}\), \(b = \frac{p q t^2 / X + q t}{u}\). Để sinh ước nhanh, chương trình lưu sẵn phân tích thừa số nguyên tố của \(p/q\) (cộng/trừ số mũ của \(a\) và \(a + t\) mỗi lần đi xuống).
    • Còn \(k \ge 3\) thừa số: thừa số đầu là lớn nhất nên \(\frac{a + t}{a} < r\) (các thừa số còn lại đều \(> 1\)) và \(\left(\frac{a + t}{a}\right)^k \ge r\). Suy ra $\(\frac{t}{r - 1} < a \le \frac{t}{r^{1/k} - 1},\)$ duyệt \(a\) trong đoạn này rồi đệ quy với \(r \cdot \frac{a}{a + t}\), \(k - 1\).

    Ví dụ \(n = 4, t = 1\): \(k = 1\) cần \(a = \frac{1}{3}\), không nguyên. \(k = 2\): \(u = 3\), \(p q t^2 = 4\), ước \(X = 2\) thỏa \(2 \equiv -1 \pmod 3\), cho \(a = b = 1\). Đáp án 2.

    Đáp án có thể khá lớn khi \(t\) nhỏ: \(n = 997, t = 1\) cần \(k = 13\), với bộ \(a = (1, 1, 1, 1, 1, 1, 1, 1, 1, 2, 4, 26, 12960)\). Các \(a_i\) cũng có thể rất lớn: \(n = 526, t = 829\) cần \(k = 4\) với \(a = (2, 3119, 6799443, 4684700580528)\). Vì vậy không thể giới hạn \(a_i\) bằng một hằng số nhỏ rồi vét cạn.

    Kiểm chứng trên toàn bộ miền dữ liệu

    Vì \(n, t \le 1000\) nên chỉ có \(10^6\) bộ dữ liệu. Nhóm biên soạn đã chạy bản C++ trên toàn bộ \(10^6\) bộ: đáp án lớn nhất là 14, mỗi bộ chạy không quá khoảng 0,1 giây, và các phép tính với __int128 không tràn ở bộ nào. Kết quả trùng với bản tìm kiếm không dùng mẹo duyệt ước (chậm hơn nhiều) trên 400 bộ ngẫu nhiên cộng toàn bộ \(n, t \le 100\).

    Bẫy thường gặp

    • Nghĩ rằng đáp án luôn là \(\lceil \log_{t+1} n \rceil\). Sai: \(n = 29, t = 58\) có \(\lceil \log_{59} 29 \rceil = 1\) nhưng đáp án là 4.
    • Giới hạn \(a_i \le\) vài nghìn rồi vét: bỏ sót nghiệm vì \(a_i\) thực tế có thể lên tới \(10^{12}\).
    • Cận trên của \(a\) tính bằng số thực: cộng thêm biên an toàn (lời giải cộng 2) để không cắt nhầm nghiệm khi \(\left(\frac{a + t}{a}\right)^k\) đúng bằng \(r\).

    Nếu viết bằng C++

    \(p\), \(q\) và \(p q t^2\) vượt long long, lời giải dùng __int128 (có sẵn trong GCC/Clang, không cần header riêng). Bản Python dùng số nguyên lớn nên không lo tràn, nhưng chậm hơn: vài bộ khó nhất (như \(n = 685, t = 821\)) mất khoảng 1 giây trên máy kiểm thử, có thể vượt giới hạn 1 giây nếu chấm như bản đề trên VNOJ; bản C++ chạy dưới 0,11 giây với mọi bộ.

    dtask.py
    import sys
from math import gcd


def phan_tich(x, mu, dau):
    # Cộng (dau = +1) hoặc trừ (dau = -1) số mũ các thừa số nguyên tố của x vào mu
    d = 2
    while d * d <= x:
        while x % d == 0:
            mu[d] = mu.get(d, 0) + dau
            x //= d
        d += 1
    if x > 1:
        mu[x] = mu.get(x, 0) + dau


def solve_dtask(data):
    n, t = map(int, data.split()[:2])
    if n == 1:
        return "-1"  # mỗi thừa số (a+t)/a > 1 nên tích luôn > 1
    tmu = {}
    phan_tich(t, tmu, 1)
    mu = {}  # số mũ nguyên tố của p/q (dương: thuộc p, âm: thuộc q)

    def hai_so(p, q, amin):
        # Còn đúng 2 thừa số, a <= b: (a*u - q*t) * (b*u - q*t) = p*q*t^2 với u = p - q.
        # Duyệt ước X <= căn(N) của N = p*q*t^2, thử X ≡ -q*t (mod u).
        u = p - q
        qt = q * t
        N = p * q * t * t
        f = {}
        for pr, e in mu.items():
            if e:
                f[pr] = f.get(pr, 0) + abs(e)
        for pr, e in tmu.items():
            f[pr] = f.get(pr, 0) + 2 * e
        ds = list(f.items())

        def gen(i, X):
            if X * X > N:
                return False
            if i == len(ds):
                if (X + qt) % u == 0 and (X + qt) // u >= amin and (N // X + qt) % u == 0:
                    return True
                return False
            pr, e = ds[i]
            for _ in range(e + 1):
                if X * X > N:
                    break
                if gen(i + 1, X):
                    return True
                X *= pr
            return False

        return gen(0, 1)

    def dfs(p, q, k, amin):
        # Viết được p/q thành tích k thừa số (a+t)/a với amin <= a1 <= a2 <= ... ?
        u = p - q
        if k == 1:
            return (t * q) % u == 0 and t * q // u >= amin
        if k == 2:
            return hai_so(p, q, amin)
        # Thừa số đầu là lớn nhất: (a+t)/a < p/q và ((a+t)/a)^k >= p/q
        lo = max(amin, t * q // u + 1)
        hi = int(t / ((p / q) ** (1.0 / k) - 1)) + 2  # cộng 2 phòng sai số số thực
        for a in range(lo, hi + 1):
            np_, nq = p * a, q * (a + t)
            g = gcd(np_, nq)
            np_ //= g
            nq //= g
            if np_ <= nq:
                continue
            phan_tich(a, mu, 1)
            phan_tich(a + t, mu, -1)
            ok = dfs(np_, nq, k - 1, a)
            phan_tich(a, mu, -1)
            phan_tich(a + t, mu, 1)
            if ok:
                return True
        return False

    # Tăng dần k (iterative deepening), k đầu tiên tìm được là nhỏ nhất
    k = 1
    while True:
        mu.clear()
        phan_tich(n, mu, 1)
        if dfs(n, 1, k, 1):
            return str(k)
        k += 1


if __name__ == "__main__":
    sys.setrecursionlimit(10000)
    try:
        with open("input.inp") as f:
            data = f.read()
        with open("output.out", "w") as f:
            f.write(solve_dtask(data) + "\n")
    except FileNotFoundError:
        print(solve_dtask(sys.stdin.read()))
    dtask.cpp
    #include <cmath>
#include <cstdio>
#include <cstdlib>
#include <iostream>
#include <map>
#include <utility>
#include <vector>
using namespace std;

typedef __int128 lll;  // p, q, p*q*t^2 có thể vượt long long

long long T;
map<long long, int> mu;                // số mũ nguyên tố của p/q (dương: thuộc p, âm: thuộc q)
vector<pair<long long, int>> tmu;      // phân tích t
vector<pair<long long, int>> ds;       // phân tích N = p*q*t^2
lll N, U, QT;
long long AMIN;

void phan_tich(long long x, int dau) {
    // Cộng (dau = +1) hoặc trừ (dau = -1) số mũ các thừa số nguyên tố của x vào mu
    for (long long d = 2; d * d <= x; d++)
        while (x % d == 0) {
            mu[d] += dau;
            x /= d;
        }
    if (x > 1) mu[x] += dau;
}

lll gcd128(lll a, lll b) {
    while (b) {
        lll c = a % b;
        a = b;
        b = c;
    }
    return a;
}

bool gen(size_t i, lll X) {
    if (X * X > N) return false;
    if (i == ds.size())
        return (X + QT) % U == 0 && (X + QT) / U >= AMIN && (N / X + QT) % U == 0;
    for (int e = 0; e <= ds[i].second; e++) {
        if (X * X > N) break;
        if (gen(i + 1, X)) return true;
        X *= ds[i].first;
    }
    return false;
}

bool hai_so(lll p, lll q, long long amin) {
    // Còn đúng 2 thừa số, a <= b: (a*u - q*t) * (b*u - q*t) = p*q*t^2 với u = p - q.
    // Duyệt ước X <= căn(N) của N, thử X ≡ -q*t (mod u).
    U = p - q;
    QT = q * T;
    N = p * q * T * T;
    AMIN = amin;
    map<long long, int> f;
    for (auto &x : mu)
        if (x.second) f[x.first] += abs(x.second);
    for (auto &x : tmu) f[x.first] += 2 * x.second;
    ds.assign(f.begin(), f.end());
    return gen(0, 1);
}

bool dfs(lll p, lll q, int k, long long amin) {
    // Viết được p/q thành tích k thừa số (a+t)/a với amin <= a1 <= a2 <= ... ?
    lll u = p - q;
    if (k == 1) return (T * q) % u == 0 && T * q / u >= amin;
    if (k == 2) return hai_so(p, q, amin);
    // Thừa số đầu là lớn nhất: (a+t)/a < p/q và ((a+t)/a)^k >= p/q
    long long lo = (long long)(T * q / u) + 1;
    if (lo < amin) lo = amin;
    long double r = (long double)p / (long double)q;
    long long hi = (long long)((long double)T / (powl(r, 1.0L / k) - 1)) + 2;  // +2 phòng sai số
    for (long long a = lo; a <= hi; a++) {
        lll np = p * a, nq = q * (a + T), g = gcd128(np, nq);
        np /= g;
        nq /= g;
        if (np <= nq) continue;
        phan_tich(a, 1);
        phan_tich(a + T, -1);
        bool ok = dfs(np, nq, k - 1, a);
        phan_tich(a, -1);
        phan_tich(a + T, 1);
        if (ok) return true;
    }
    return false;
}

int solve_dtask(int n, int t) {
    if (n == 1) return -1;  // mỗi thừa số (a+t)/a > 1 nên tích luôn > 1
    T = t;
    tmu.clear();
    long long x = t;
    for (long long d = 2; d * d <= x; d++) {
        int c = 0;
        while (x % d == 0) {
            c++;
            x /= d;
        }
        if (c) tmu.push_back({d, c});
    }
    if (x > 1) tmu.push_back({x, 1});
    // Tăng dần k (iterative deepening), k đầu tiên tìm được là nhỏ nhất
    for (int k = 1;; k++) {
        mu.clear();
        phan_tich(n, 1);
        if (dfs(n, 1, k, 1)) return k;
    }
}

    // Có input.inp thì đọc/ghi file, không thì dùng bàn phím/màn hình

    }

}

    3. Xâu con¶

    Ý tưởng. \(r(s)^2\) là số cặp cách chọn \((I, J)\) (mỗi cách chọn là một dãy chỉ số tăng) cùng cho ra xâu con \(s\). Vậy

    \[f(S) = \sum_{\text{cặp } (I, J) \text{ cùng cho một xâu con}} |I|.\]

    Hai dãy chỉ số \(i_1 < \ldots < i_\ell\) và \(j_1 < \ldots < j_\ell\) cho cùng một xâu khi \(S[i_x] = S[j_x]\) với mọi \(x\). Đếm theo cặp vị trí cuối cùng \((i, j)\) với \(S[i] = S[j]\):

    • \(cnt[i][j] = 1 + \sum_{i' < i,\ j' < j} cnt[i'][j']\) (cặp chỉ gồm \((i, j)\), hoặc nối \((i, j)\) vào sau một cặp trước đó);
    • \(len[i][j] = cnt[i][j] + \sum_{i' < i,\ j' < j} len[i'][j']\) (mỗi cặp được nối dài thêm 1).

    Đáp án là \(\sum len[i][j]\). Hai tổng \(\sum_{i' < i, j' < j}\) lấy bằng tổng tiền tố 2 chiều nên cả bài là \(O(N^2)\), với \(N = 200\) chỉ \(4 \cdot 10^4\) ô.

    Kiểm tra với \(S = ABAB\): chương trình in 48, khớp ví dụ.

    Bẫy thường gặp

    • Sinh tất cả \(2^N - 1\) xâu con chỉ qua được subtask \(|S| \le 20\).
    • Trừ trong tổng tiền tố có thể ra số âm khi lấy modulo; C++ phải cộng thêm MOD trước khi %.
    subseq.py
    import sys

MOD = 10**9 + 7


def solve_subseq(data):
    s = data.split()[0] if data.split() else ""
    n = len(s)
    # Σ r(s)^2 * |s| = tổng độ dài trên mọi CẶP cách chọn (I, J) cho ra cùng một xâu con.
    # cnt[i][j]: số cặp kết thúc đúng tại (i, j) với s[i] == s[j]; ln[i][j]: tổng độ dài của chúng.
    # P, Q: tổng tiền tố 2 chiều của cnt, ln (chỉ số lệch 1).
    P = [[0] * (n + 1) for _ in range(n + 1)]
    Q = [[0] * (n + 1) for _ in range(n + 1)]
    ans = 0
    for i in range(1, n + 1):
        Pi, Pp, Qi, Qp = P[i], P[i - 1], Q[i], Q[i - 1]
        ci = s[i - 1]
        for j in range(1, n + 1):
            cnt = ln = 0
            if s[j - 1] == ci:
                prev_c = Pp[j - 1]  # tổng cnt của các cặp (i', j') với i' < i, j' < j
                cnt = (1 + prev_c) % MOD  # bắt đầu mới, hoặc nối dài một cặp trước đó
                ln = (cnt + Qp[j - 1]) % MOD  # mỗi cặp nối dài thêm 1 ký tự
                ans += ln
            Pi[j] = (cnt + Pp[j] + Pi[j - 1] - Pp[j - 1]) % MOD
            Qi[j] = (ln + Qp[j] + Qi[j - 1] - Qp[j - 1]) % MOD
    return str(ans % MOD)

    #include <cstdio>
#include <iostream>
#include <string>
#include <vector>
using namespace std;

const long long MOD = 1000000007LL;

long long solve_subseq(const string &s) {
    int n = s.size();
    // Σ r(s)^2 * |s| = tổng độ dài trên mọi CẶP cách chọn (I, J) cho ra cùng một xâu con.
    // cnt: số cặp kết thúc đúng tại (i, j) với s[i] == s[j]; ln: tổng độ dài của chúng.
    // P, Q: tổng tiền tố 2 chiều của cnt, ln (chỉ số lệch 1).
    vector<vector<long long>> P(n + 1, vector<long long>(n + 1, 0)), Q = P;
    long long ans = 0;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++) {
            long long cnt = 0, ln = 0;
            if (s[j - 1] == s[i - 1]) {
                cnt = (1 + P[i - 1][j - 1]) % MOD;  // bắt đầu mới, hoặc nối dài cặp trước
                ln = (cnt + Q[i - 1][j - 1]) % MOD;  // mỗi cặp nối dài thêm 1 ký tự
                ans = (ans + ln) % MOD;
            }
            P[i][j] = ((cnt + P[i - 1][j] + P[i][j - 1] - P[i - 1][j - 1]) % MOD + MOD) % MOD;
            Q[i][j] = ((ln + Q[i - 1][j] + Q[i][j - 1] - Q[i - 1][j - 1]) % MOD + MOD) % MOD;
        }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    // Có input.inp thì đọc/ghi file, không thì dùng bàn phím/màn hình

        fclose(f);
        freopen("input.inp", "r", stdin);
        freopen("output.out", "w", stdout);
    }
    string s;
    cin >> s;
    cout << solve_subseq(s) << "\n";
    return 0;
}

    4. Cắt bánh¶

    Ý tưởng. Bỏ các lần cắt trùng nhau và các lần cắt dọc theo cạnh bánh (hai đỉnh kề nhau), còn \(D \le 1225\) đường cắt khác nhau. Thêm lần lượt từng đường cắt: nếu đường mới cắt các đường cũ tại \(x\) điểm phân biệt bên trong bánh, nó đi qua \(x + 1\) miếng và chia đôi mỗi miếng, nên số miếng tăng \(x + 1\). Đáp án là \(1 + \sum (x + 1)\).

    • Hai dây cung \((i, j)\) và \((a, b)\) cắt nhau bên trong bánh khi và chỉ khi đầu mút xen kẽ: đúng một trong \(a, b\) nằm giữa \(i\) và \(j\) (theo thứ tự đỉnh), và không chung đỉnh. Phần này dùng số nguyên nên chính xác.
    • Để biết các giao điểm có trùng nhau không (nhiều đường cắt đồng quy, chuyện thường gặp ở đa giác đều), tính vị trí giao điểm trên đường mới dưới dạng tham số \(t \in (0, 1)\), sắp xếp và đếm các giá trị khác nhau với sai số \(10^{-9}\).

    Ví dụ: hình vuông cắt theo đường chéo \(1\)–\(3\) thì \(x = 0\), số miếng \(1 + 1 = 2\).

    Vì sao sai số \(10^{-9}\) là an toàn

    Nhóm biên soạn đã chạy cho mọi \(n \le 50\) với tất cả đường chéo: hai giao điểm trùng nhau lệch nhau tối đa khoảng \(4{,}4 \cdot 10^{-15}\) trên tham số \(t\), còn hai giao điểm khác nhau cách nhau ít nhất khoảng \(2{,}6 \cdot 10^{-6}\). Tập con đường cắt bất kỳ chỉ có ít giao điểm hơn nên khoảng cách này vẫn đúng. Số miếng khi cắt mọi đường chéo khớp dãy OEIS A007678 với mọi \(3 \le n \le 50\) (ví dụ \(n = 50\) cho 220600 miếng).

    Bẫy thường gặp

    • Dùng công thức \(1 + D + (\text{số cặp đường cắt nhau})\): chỉ đúng với 30% test không có ba đường đồng quy. Đa giác đều có rất nhiều điểm đồng quy (ví dụ ba đường chéo chính của lục giác đều).
    • Quên bỏ lần cắt trùng nhau hoặc ghi ngược (3 1 và 1 3), hoặc lần cắt dọc theo cạnh.
    • Đếm cả giao điểm tại đỉnh của bánh.
    cutcake.py
    import sys
from math import cos, sin, pi

EPS = 1e-9


def solve_cutcake(data):
    tokens = data.split()
    n, m = int(tokens[0]), int(tokens[1])
    # Gom các đường cắt khác nhau; bỏ cạnh của đa giác (hai đỉnh kề nhau không cắt gì)
    seen = set()
    chords = []
    for k in range(m):
        i, j = int(tokens[2 + 2 * k]) - 1, int(tokens[3 + 2 * k]) - 1
        if i > j:
            i, j = j, i
        if (i, j) in seen or j - i == 1 or j - i == n - 1:
            continue
        seen.add((i, j))
        chords.append((i, j))
    xs = [cos(2 * pi * v / n) for v in range(n)]
    ys = [sin(2 * pi * v / n) for v in range(n)]

    # Thêm lần lượt từng đường cắt: số miếng tăng 1 + (số điểm PHÂN BIỆT
    # bên trong bánh mà nó cắt các đường trước đó).
    pieces = 1
    for idx, (i, j) in enumerate(chords):
        x1, y1 = xs[i], ys[i]
        dx, dy = xs[j] - x1, ys[j] - y1
        ts = []
        for (a, b) in chords[:idx]:
            # Hai dây cung cắt nhau bên trong <=> đầu mút xen kẽ nhau
            if (i < a < j) == (i < b < j) or a in (i, j) or b in (i, j):
                continue
            ex, ey = xs[b] - xs[a], ys[b] - ys[a]
            den = dx * ey - dy * ex
            tt = ((xs[a] - x1) * ey - (ys[a] - y1) * ex) / den  # vị trí giao điểm trên dây (i, j)
            ts.append(tt)
        ts.sort()
        distinct = 0
        last = -1.0
        for tt in ts:
            if tt - last > EPS:
                distinct += 1
                last = tt
        pieces += 1 + distinct
    return str(pieces)

    #include <algorithm>
#include <cmath>
#include <cstdio>
#include <iostream>
#include <set>
#include <utility>
#include <vector>
using namespace std;

const double EPS = 1e-9;

long long solve_cutcake(int n, const vector<pair<int, int>> &cuts) {
    // Gom các đường cắt khác nhau; bỏ cạnh của đa giác (hai đỉnh kề nhau không cắt gì)
    set<pair<int, int>> seen;
    vector<pair<int, int>> chords;
    for (auto c : cuts) {
        int i = c.first - 1, j = c.second - 1;
        if (i > j) swap(i, j);
        if (seen.count({i, j}) || j - i == 1 || j - i == n - 1) continue;
        seen.insert({i, j});
        chords.push_back({i, j});
    }
    const double PI = acos(-1.0);
    vector<double> xs(n), ys(n);
    for (int v = 0; v < n; v++) {
        xs[v] = cos(2 * PI * v / n);
        ys[v] = sin(2 * PI * v / n);
    }
    // Thêm lần lượt từng đường cắt: số miếng tăng 1 + (số điểm PHÂN BIỆT
    // bên trong bánh mà nó cắt các đường trước đó).
    long long pieces = 1;
    for (size_t idx = 0; idx < chords.size(); idx++) {
        int i = chords[idx].first, j = chords[idx].second;
        double x1 = xs[i], y1 = ys[i], dx = xs[j] - x1, dy = ys[j] - y1;
        vector<double> ts;
        for (size_t o = 0; o < idx; o++) {
            int a = chords[o].first, b = chords[o].second;
            // Hai dây cung cắt nhau bên trong <=> đầu mút xen kẽ nhau
            if ((i < a && a < j) == (i < b && b < j) || a == i || a == j || b == i || b == j) continue;
            double ex = xs[b] - xs[a], ey = ys[b] - ys[a];
            double den = dx * ey - dy * ex;
            ts.push_back(((xs[a] - x1) * ey - (ys[a] - y1) * ex) / den);  // vị trí giao điểm
        }
        sort(ts.begin(), ts.end());
        long long distinct = 0;
        double last = -1.0;
        for (double tt : ts)
            if (tt - last > EPS) {
                distinct++;
                last = tt;
            }
        pieces += 1 + distinct;
    }
    return pieces;
}
