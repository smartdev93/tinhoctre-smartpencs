    Đáp án - Bảng B Quận Hải Châu 2025¶

    Lời giải tham khảo bằng Python 3 và C++ (C++17). Xem lại đề bài.

    Mỗi chương trình đọc từ bàn phím và in ra màn hình như code mẫu của tài liệu. Nếu có file input.inp, chương trình đọc từ file và ghi kết quả ra output.out.

    STT Bài Ý tưởng Độ phức tạp
    1 Cặp số thân thiện Tính tổng ước bằng cách duyệt đến \(\sqrt{N}\), kiểm tra \(A \ne B\) \(O(\sqrt{N})\)
    2 Nén xâu đảo ngược Đếm từng đoạn ký tự giống nhau, rồi đảo xâu \(O(N)\)
    3 Đoạn con có tổng lớn nhất Kadane \(O(N)\)
    4 Sắp đặt hàng hóa Sắp xếp + hai con trỏ \(O(N \log N)\)

    1. Cặp số thân thiện¶

    Ý tưởng. Ước của \(n\) đi theo cặp \((i, n/i)\) với \(i \le \sqrt{n}\). Duyệt \(i\) từ 2 đến \(\sqrt{n}\), mỗi lần \(i\) chia hết \(n\) thì cộng cả \(i\) và \(n/i\) (chỉ cộng một lần nếu \(i = n/i\)). Cộng thêm ước 1, không cộng chính \(n\).

    Với \(N = 10^{12}\), vòng lặp chạy khoảng \(10^6\) bước thay vì \(10^{12}\).

    Đáp án là YES khi \(A \ne B\), tổng ước thực sự của \(A\) bằng \(B\) và tổng ước thực sự của \(B\) bằng \(A\). Lời giải kiểm tra \(A \ne B\) trước, và chỉ tính tổng ước của \(B\) khi tổng ước của \(A\) đã bằng \(B\), nên phần lớn test chỉ tốn một vòng \(\sqrt{N}\).

    Bẫy thường gặp

    • Quên điều kiện \(A \ne B\): test 6 6 (số hoàn hảo) sẽ in sai YES.
    • \(n = 1\) không có ước thực sự nào, tổng là 0. Nếu duyệt từ \(i = 1\) và cộng cả \(i\) lẫn \(n / i\) mà không xử lý riêng, tổng ước của 1 thành 1. Code mẫu trong tài liệu Tổng hợp đề thi & đáp án mắc lỗi này. Lời giải này trả về 0 khi \(n \le 1\).
    • Duyệt đến \(n - 1\) hoặc \(n / 2\) quá chậm với \(N = 10^{12}\).
    • Điều kiện dừng i * i <= n tránh dùng sqrt số thực (có thể sai số với số lớn).

    Nếu viết bằng C++

    \(A, B\) và tổng ước có thể vượt \(2^{31}\), phải dùng long long cho cả biến chạy i (vì i * i lên tới \(10^{12}\)).

    friend.py
    import sys


def tong_uoc_thuc_su(n):
    # Tổng các ước của n, không kể chính n. Duyệt i đến căn n -> O(√n).
    if n <= 1:
        return 0
    s = 1
    i = 2
    while i * i <= n:
        if n % i == 0:
            s += i
            if i * i != n:
                s += n // i  # ước đối xứng n / i
        i += 1
    return s


def solve_friend(data):
    a, b = map(int, data.split()[:2])
    # Cặp thân thiện phải gồm hai số khác nhau (6 6 là số hoàn hảo -> NO).
    # Kiểm tra A trước, chỉ tính tổng ước của B khi cần -> đỡ một lần O(√N)
    if a != b and tong_uoc_thuc_su(a) == b and tong_uoc_thuc_su(b) == a:
        return "YES"
    return "NO"


if __name__ == "__main__":
    try:
        with open("input.inp") as f:
            data = f.read()
        with open("output.out", "w") as f:
            f.write(solve_friend(data) + "\n")
    except FileNotFoundError:
        print(solve_friend(sys.stdin.read()))
    friend.cpp
    #include <cstdio>
#include <iostream>
#include <string>
using namespace std;

long long tong_uoc_thuc_su(long long n) {
    // Tổng các ước của n, không kể chính n. Duyệt i đến căn n -> O(√n).
    if (n <= 1) return 0;
    long long s = 1;
    for (long long i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            s += i;
            if (i * i != n) s += n / i;  // ước đối xứng n / i
        }
    }
    return s;
}

string solve_friend(long long a, long long b) {
    // Cặp thân thiện phải gồm hai số khác nhau (6 6 là số hoàn hảo -> NO).
    // Kiểm tra A trước, chỉ tính tổng ước của B khi cần -> đỡ một lần O(√N)
    if (a != b && tong_uoc_thuc_su(a) == b && tong_uoc_thuc_su(b) == a) return "YES";
    return "NO";
}

int main() {
    if (FILE *f = fopen("input.inp", "r")) {
        fclose(f);
        freopen("input.inp", "r", stdin);
        freopen("output.out", "w", stdout);
    }
    long long a, b;  // A, B tới 10^12 -> long long
    cin >> a >> b;
    cout << solve_friend(a, b) << "\n";
    return 0;
}

    2. Nén xâu đảo ngược¶

    Ý tưởng. Duyệt xâu, tại mỗi vị trí \(i\) tìm \(j\) là vị trí đầu tiên có ký tự khác \(S[i]\). Đoạn \(S[i..j-1]\) nén thành str(j - i) + S[i]. Nhảy \(i = j\) và lặp lại. Cuối cùng đảo ngược cả xâu kết quả.

    Mỗi ký tự được duyệt đúng một lần nên độ phức tạp \(O(N)\).

    Bẫy thường gặp

    • Nối xâu bằng res = res + ... trong vòng lặp có thể thành \(O(N^2)\) trong một số ngôn ngữ. Python nên gom vào danh sách rồi "".join(...); C++ dùng res += ... là ổn.
    • Số lần lặp có thể có nhiều chữ số (12a). Đề yêu cầu đảo toàn bộ xâu nên chữ số cũng bị đảo (a21), không đảo theo từng cụm.
    compress.py
    import sys


def solve_compress(data):
    s = data.strip()
    n = len(s)
    parts = []
    i = 0
    while i < n:
        # Đếm đoạn ký tự giống nhau bắt đầu tại i
        j = i
        while j < n and s[j] == s[i]:
            j += 1
        parts.append(str(j - i) + s[i])
        i = j
    # Nối bằng join (O(N)), rồi đảo ngược cả xâu kết quả
    return "".join(parts)[::-1]


if __name__ == "__main__":
    try:
        with open("input.inp") as f:
            data = f.read()
        with open("output.out", "w") as f:
            f.write(solve_compress(data) + "\n")
    except FileNotFoundError:
        print(solve_compress(sys.stdin.read()))
    compress.cpp
    #include <algorithm>
#include <cstdio>
#include <iostream>
#include <iterator>
#include <string>
using namespace std;

string solve_compress(const string &data) {
    // Bỏ khoảng trắng / xuống dòng ở hai đầu (giống strip() của Python)
    const string ws = " \t\r\n\v\f";
    size_t l = data.find_first_not_of(ws);
    if (l == string::npos) return "";
    size_t r = data.find_last_not_of(ws);
    string s = data.substr(l, r - l + 1);

    int n = s.size();
    string res;
    int i = 0;
    while (i < n) {
        // Đếm đoạn ký tự giống nhau bắt đầu tại i
        int j = i;
        while (j < n && s[j] == s[i]) j++;
        res += to_string(j - i);
        res += s[i];
        i = j;
    }
    // Nối bằng += (O(N)), rồi đảo ngược cả xâu kết quả
    reverse(res.begin(), res.end());
    return res;
}

int main() {
    if (FILE *f = fopen("input.inp", "r")) {
        fclose(f);
        freopen("input.inp", "r", stdin);
        freopen("output.out", "w", stdout);
    }
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    string data((istreambuf_iterator<char>(cin)), istreambuf_iterator<char>());
    cout << solve_compress(data) << "\n";
    return 0;
}

    3. Đoạn con có tổng lớn nhất¶

    Ý tưởng (Kadane). Gọi \(cur\) là tổng lớn nhất của đoạn con kết thúc tại phần tử đang xét \(x\). Có hai lựa chọn: nối \(x\) vào đoạn tốt nhất kết thúc ở phần tử trước, hoặc bắt đầu đoạn mới chỉ gồm \(x\):

    \[cur = \max(x,\ cur + x)\]

    Đáp án là giá trị lớn nhất của \(cur\) trong quá trình duyệt.

    Bẫy thường gặp

    • Khởi tạo đáp án bằng 0 sẽ sai khi mọi phần tử đều âm (đáp án đúng là phần tử lớn nhất, không phải 0).
    • Vét cạn mọi đoạn \(O(N^2)\) hoặc \(O(N^3)\) quá chậm với \(N = 10^5\).

    Nếu viết bằng C++

    Tổng có thể tới \(10^5 \cdot 10^9 = 10^{14}\), phải dùng long long.

    maxsub.py
    import sys


def solve_maxsub(data):
    tokens = data.split()
    n = int(tokens[0])
    a = map(int, tokens[1:n + 1])

    # Kadane: cur = tổng lớn nhất của đoạn con kết thúc tại phần tử đang xét
    best = None
    cur = 0
    for x in a:
        cur = max(x, cur + x)  # nối tiếp đoạn cũ hoặc bắt đầu đoạn mới tại x
        if best is None or cur > best:
            best = cur
    return str(best)


if __name__ == "__main__":
    try:
        with open("input.inp") as f:
            data = f.read()
        with open("output.out", "w") as f:
            f.write(solve_maxsub(data) + "\n")
    except FileNotFoundError:
        print(solve_maxsub(sys.stdin.read()))
    maxsub.cpp
    #include <algorithm>
#include <cstdio>
#include <iostream>
using namespace std;

int main() {
    if (FILE *f = fopen("input.inp", "r")) {
        fclose(f);
        freopen("input.inp", "r", stdin);
        freopen("output.out", "w", stdout);
    }
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;

    // Kadane: cur = tổng lớn nhất của đoạn con kết thúc tại phần tử đang xét
    // Tổng tới 10^5 * 10^9 = 10^14 -> long long
    long long best = 0, cur = 0;
    for (int i = 0; i < n; i++) {
        long long x;
        cin >> x;
        cur = max(x, cur + x);  // nối tiếp đoạn cũ hoặc bắt đầu đoạn mới tại x
        if (i == 0 || cur > best) best = cur;
    }
    cout << best << "\n";
    return 0;
}

    4. Sắp đặt hàng hóa¶

    Ý tưởng. Sắp xếp các kiện tăng dần. Kiện nặng nhất (con trỏ right) chắc chắn chiếm một thùng. Nó nên đi cùng kiện nhẹ nhất (con trỏ left) nếu vừa: nếu kiện nhẹ nhất không ghép được với kiện nặng nhất thì không kiện nào ghép được, còn nếu ghép được thì chọn kiện nhẹ nhất là an toàn nhất (để dành các kiện nặng hơn cho những kiện còn lại, vốn nhẹ hơn kiện nặng nhất).

    Mỗi bước, right giảm 1 và số thùng tăng 1; nếu ghép được thì left tăng 1.

    Lưu ý về giả định của đề

    Thuật toán này chỉ đúng khi mỗi thùng chứa tối đa 2 kiện, đúng như giả định bổ sung của tài liệu. Lời giải đã được đối chiếu với vét cạn trên 400 test ngẫu nhiên nhỏ theo giả định đó.

    Nếu viết bằng C++

    \(W_{left} + W_{right}\) có thể tới \(2 \cdot 10^9\), vượt int; lời giải lưu trọng lượng bằng long long.

    arrange.py
    import sys


def solve_arrange(data):
    tokens = data.split()
    n, k = int(tokens[0]), int(tokens[1])
    w = sorted(map(int, tokens[2:2 + n]))

    # Hai con trỏ: kiện nặng nhất (right) luôn chiếm một thùng.
    # Nếu ghép được với kiện nhẹ nhất (left) thì cho vào chung thùng.
    left, right = 0, n - 1
    ans = 0
    while left <= right:
        if left < right and w[left] + w[right] <= k:
            left += 1
        right -= 1
        ans += 1
    return str(ans)


if __name__ == "__main__":
    try:
        with open("input.inp") as f:
            data = f.read()
        with open("output.out", "w") as f:
            f.write(solve_arrange(data) + "\n")
    except FileNotFoundError:
        print(solve_arrange(sys.stdin.read()))
    arrange.cpp
    #include <algorithm>
#include <cstdio>
#include <iostream>
#include <vector>
using namespace std;

int solve_arrange(vector<long long> w, long long k) {
    sort(w.begin(), w.end());
    // Hai con trỏ: kiện nặng nhất (right) luôn chiếm một thùng.
    // Nếu ghép được với kiện nhẹ nhất (left) thì cho vào chung thùng.
    int left = 0, right = (int)w.size() - 1;
    int ans = 0;
    while (left <= right) {
        if (left < right && w[left] + w[right] <= k) left++;  // tổng tới 2*10^9 -> long long
        right--;
        ans++;
    }
    return ans;
}

int main() {
    if (FILE *f = fopen("input.inp", "r")) {
        fclose(f);
        freopen("input.inp", "r", stdin);
        freopen("output.out", "w", stdout);
    }
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    long long k;
    cin >> n >> k;
    vector<long long> w(n);
    for (auto &x : w) cin >> x;
    cout << solve_arrange(w, k) << "\n";
    return 0;
}
