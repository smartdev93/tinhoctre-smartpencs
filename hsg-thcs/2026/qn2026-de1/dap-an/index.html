    Đáp án - HSG Tin học THCS Quảng Ninh 2026¶

    Lời giải tham khảo bằng Python 3 và C++ (C++17). Xem lại đề bài.

    Mỗi chương trình đọc và ghi đúng tên tệp đề yêu cầu (sal.inp / sal.out, val.inp / val.out, ...). Nếu không có tệp dữ liệu vào, chương trình đọc từ bàn phím và in ra màn hình để tiện thử.

    Bài Tên bài Ý tưởng Độ phức tạp
    1 Tiền công Tính theo số tuần trọn vẹn, cộng thêm tối đa 6 ngày lẻ \(O(1)\)
    2 Giá trị Hai con trỏ, đếm số giá trị khác nhau trong cửa sổ \(O(n)\)
    3 Dãy số Quy hoạch động cái túi hai chiều, mỗi phần tử có 4 lựa chọn \(O(n \cdot k_1 \cdot k_2)\)
    4 Số cặp Với mỗi \(i\), \(\max a - \min b\) không giảm theo \(j\); hai con trỏ + hàng đợi đơn điệu \(O(n)\)

    1. Tiền công¶

    Ý tưởng. Mỗi tuần trọn vẹn (7 ngày liên tiếp, bắt đầu từ thứ nào cũng vậy) có đúng 5 ngày thường và 2 ngày cuối tuần, nên \(\lfloor d / 7 \rfloor\) tuần đầu cho \(\lfloor d / 7 \rfloor \cdot (5a + 2b)\) đồng. Còn lại \(d \bmod 7 \le 6\) ngày, duyệt từng ngày bắt đầu từ thứ \(k\): thứ 7 (thứ Bảy) và 8 (Chủ Nhật) được \(b\), còn lại được \(a\); sau thứ 8 quay về thứ 2.

    Ví dụ 4: \(k = 3\), \(d = 12\). Một tuần trọn: \(5 \cdot 1 + 2 \cdot 2 = 9\). Còn 5 ngày từ thứ Ba đến thứ Bảy: \(4 \cdot 1 + 2 = 6\). Tổng \(15\).

    Bẫy thường gặp

    • Quên quay vòng sau Chủ Nhật (\(k = 8\) thì ngày kế tiếp là thứ Hai, mã 2).
    • Chỉ làm đúng ràng buộc 1 (\(k = 2\), \(d \le 5\), toàn ngày thường, đáp án \(d \cdot a\)) rồi quên trường hợp tổng quát.
    sal.py
    import sys


def solve_sal(data):
    k, d, a, b = map(int, data.split()[:4])
    # Mỗi tuần trọn vẹn có 5 ngày thường và 2 ngày cuối tuần
    tong = (d // 7) * (5 * a + 2 * b)
    # Duyệt các ngày lẻ còn lại (tối đa 6 ngày)
    thu = k
    for _ in range(d % 7):
        tong += b if thu >= 7 else a  # 7 = thứ Bảy, 8 = Chủ Nhật
        thu = thu + 1 if thu < 8 else 2
    return str(tong)


if __name__ == "__main__":
    try:
        with open("sal.inp") as f:
            data = f.read()
        with open("sal.out", "w") as f:
            f.write(solve_sal(data) + "\n")
    except FileNotFoundError:
        print(solve_sal(sys.stdin.read()))
    sal.cpp
    #include <cstdio>
#include <iostream>
using namespace std;

long long solve_sal(long long k, long long d, long long a, long long b) {
    // Mỗi tuần trọn vẹn có 5 ngày thường và 2 ngày cuối tuần
    long long tong = (d / 7) * (5 * a + 2 * b);
    // Duyệt các ngày lẻ còn lại (tối đa 6 ngày)
    long long thu = k;
    for (long long i = 0; i < d % 7; i++) {
        tong += (thu >= 7) ? b : a;  // 7 = thứ Bảy, 8 = Chủ Nhật
        thu = (thu < 8) ? thu + 1 : 2;
    }
    return tong;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    // Có sal.inp thì đọc/ghi file, không thì dùng bàn phím/màn hình
    if (FILE *f = fopen("sal.inp", "r")) {
        fclose(f);
        freopen("sal.inp", "r", stdin);
        freopen("sal.out", "w", stdout);
    }
    long long k, d, a, b;
    cin >> k >> d >> a >> b;
    cout << solve_sal(k, d, a, b) << "\n";
    return 0;
}

    2. Giá trị¶

    Ý tưởng. Hai con trỏ (cửa sổ trượt). Duyệt đầu phải \(r\), thêm \(a_r\) vào cửa sổ và cập nhật dem[v] (số lần giá trị \(v\) xuất hiện), khac (số giá trị khác nhau). Khi khac > k thì dịch đầu trái \(l\) sang phải cho đến khi khac \(\le k\). Lúc này \([l, r]\) là đoạn dài nhất kết thúc tại \(r\) có không quá \(k\) giá trị; nếu khac == k thì cập nhật đáp án.

    Vì \(l\) chỉ tăng, mỗi phần tử vào và ra cửa sổ tối đa một lần: \(O(n)\).

    Với \(k = 1\), bài toán chính là tìm đoạn dài nhất gồm các số bằng nhau. Ràng buộc 1 (dãy không giảm, chỉ có 1 và 2) có thể đếm số lượng số 1 và số 2 rồi lấy max.

    Bẫy thường gặp

    • "Đúng \(k\) giá trị" chứ không phải "không quá \(k\)": với \(k = 2\), đoạn chỉ gồm một giá trị không được tính. Lời giải chỉ cập nhật khi khac == k.
    • Duyệt mọi cặp \((i, j)\) là \(O(n^2) = 10^{10}\) phép, quá chậm với \(n = 10^5\).
    val.py
    import sys


def solve_val(data):
    t = list(map(int, data.split()))
    n, k = t[0], t[1]
    a = t[2:2 + n]
    dem = [0] * 4  # dem[v] = số lần giá trị v xuất hiện trong cửa sổ
    khac = 0       # số giá trị khác nhau trong cửa sổ a[l..r]
    l = 0
    best = 0
    for r in range(n):
        if dem[a[r]] == 0:
            khac += 1
        dem[a[r]] += 1
        # Thu hẹp bên trái khi cửa sổ có hơn k giá trị
        while khac > k:
            dem[a[l]] -= 1
            if dem[a[l]] == 0:
                khac -= 1
            l += 1
        if khac == k and r - l + 1 > best:
            best = r - l + 1
    return str(best)


if __name__ == "__main__":
    try:
        with open("val.inp") as f:
            data = f.read()
        with open("val.out", "w") as f:
            f.write(solve_val(data) + "\n")
    except FileNotFoundError:
        print(solve_val(sys.stdin.read()))
    val.cpp
    #include <cstdio>
#include <iostream>
#include <vector>
using namespace std;

int solve_val(int k, const vector<int> &a) {
    int n = a.size();
    int dem[4] = {0, 0, 0, 0};  // dem[v] = số lần giá trị v xuất hiện trong cửa sổ
    int khac = 0;               // số giá trị khác nhau trong cửa sổ a[l..r]
    int l = 0, best = 0;
    for (int r = 0; r < n; r++) {
        if (dem[a[r]] == 0) khac++;
        dem[a[r]]++;
        // Thu hẹp bên trái khi cửa sổ có hơn k giá trị
        while (khac > k) {
            dem[a[l]]--;
            if (dem[a[l]] == 0) khac--;
            l++;
        }
        if (khac == k && r - l + 1 > best) best = r - l + 1;
    }
    return best;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    // Có val.inp thì đọc/ghi file, không thì dùng bàn phím/màn hình
    if (FILE *f = fopen("val.inp", "r")) {
        fclose(f);
        freopen("val.inp", "r", stdin);
        freopen("val.out", "w", stdout);
    }
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    cout << solve_val(k, a) << "\n";
    return 0;
}

    3. Dãy số¶

    Ý tưởng. Mỗi phần tử \(x\) chỉ có 4 kết cục, không phụ thuộc phần tử khác:

    Lựa chọn Giá trị mới Tốn
    Không làm gì \(x\) 0 phép
    Chỉ loại 1 \(\lfloor x / 2 \rfloor\) 1 phép loại 1
    Chỉ loại 2 \(\max(x - b, 0)\) 1 phép loại 2
    Cả hai \(\min\big(\max(\lfloor x/2 \rfloor - b, 0),\ \lfloor \max(x - b, 0) / 2 \rfloor\big)\) 1 phép mỗi loại

    Ràng buộc duy nhất giữa các phần tử là tổng số phép mỗi loại, nên đây là bài cái túi hai chiều. Gọi \(dp[j_1][j_2]\) là lượng giảm lớn nhất khi dùng tối đa \(j_1\) phép loại 1 và \(j_2\) phép loại 2 trên các phần tử đã xét. Thêm phần tử \(x\) với lượng giảm \(g_1, g_2, g_{12}\) của ba lựa chọn có làm gì đó:

    \[ dp[j_1][j_2] = \max\big(dp[j_1][j_2],\ dp[j_1][j_2-1] + g_2,\ dp[j_1-1][j_2] + g_1,\ dp[j_1-1][j_2-1] + g_{12}\big) \]

    (vế phải dùng giá trị trước khi thêm \(x\)). Đáp án là \(\sum a_i - dp[k_1][k_2]\). Khởi tạo toàn 0 vì "tối đa" cho phép không dùng hết số phép.

    Với hai loại trên cùng một số, chia đôi trước rồi trừ luôn không tệ hơn (vì \(b \ge 1\)), nhưng lời giải lấy \(\min\) của hai thứ tự cho chắc. Trong ví dụ 3, \(a_1 = 19\): chia đôi trước \(\to 9 \to 0\); trừ trước \(\to 10 \to 5\).

    Vì sao không tham lam: một phần tử lớn có thể đáng dùng cả hai phép, hoặc đáng để dành phép loại 1 cho số khác; thứ tự ưu tiên giữa hai loại phụ thuộc vào \(b\) và cả dãy, nên phải xét hết bằng quy hoạch động.

    Độ phức tạp \(O(n \cdot k_1 \cdot k_2) \le 300^3 = 2{,}7 \cdot 10^7\), C++ chạy dưới 0,1 giây.

    Bản Python cần thêm hai tối ưu để kịp 1 giây:

    • Chỉ cập nhật các ô còn có thể dẫn tới \(dp[k_1][k_2]\): sau phần tử thứ \(i\) (đếm từ 0) còn \(n - 1 - i\) phần tử, mỗi phần tử tăng \(j_1\) tối đa 1, nên chỉ cần \(j_1 \ge k_1 - (n - 1 - i)\); đồng thời \(j_1 \le i + 1\). Tương tự với \(j_2\). Số ô thực sự cập nhật giảm mạnh (khi \(k_1 = k_2 = n\) chỉ còn 1 ô mỗi phần tử; trường hợp nặng nhất là khoảng \(k_1 = k_2 = n/2\)).
    • Cập nhật cả một hàng bằng list comprehension (tính từ giá trị cũ rồi gán một lần), duyệt \(j_1\) giảm dần để hàng \(j_1 - 1\) vẫn là giá trị cũ.

    Thử trên máy biên soạn với \(n = 300\) và nhiều cặp \((k_1, k_2)\): bản Python chạy tối đa khoảng 0,45 giây (trường hợp \(k_1 = k_2 = 150\)), bản C++ khoảng 0,02 giây.

    Bẫy thường gặp

    • Quên lựa chọn "cả hai phép trên cùng một số" (ví dụ 3 cần dùng cả hai phép cho \(a_1\)).
    • Áp dụng một phép nhiều lần lên cùng một số: đề chỉ cho mỗi loại tối đa 1 lần trên mỗi \(a_i\).
    • Duyệt \(j_1, j_2\) tăng dần khi dùng mảng hai chiều: một phần tử bị dùng nhiều lần.

    Nếu viết bằng C++

    Tổng dãy lên tới \(300 \cdot 10^9 = 3 \cdot 10^{11}\), vượt int: dùng long long cho \(a_i\), \(b\), tổng và mảng \(dp\).

    arr.py
    import sys


def solve_arr(data):
    t = list(map(int, data.split()))
    n, b, k1, k2 = t[0], t[1], t[2], t[3]
    a = t[4:4 + n]
    k1 = min(k1, n)
    k2 = min(k2, n)
    # dp[j1][j2] = lượng giảm lớn nhất khi dùng tối đa j1 phép loại 1, j2 phép loại 2
    dp = [[0] * (k2 + 1) for _ in range(k1 + 1)]
    for i, x in enumerate(a):
        v1 = x // 2                      # chỉ loại 1
        v2 = max(x - b, 0)               # chỉ loại 2
        v12 = min(max(v1 - b, 0), v2 // 2)  # cả hai, chọn thứ tự tốt hơn
        g1, g2, g12 = x - v1, x - v2, x - v12
        # Chỉ cần các ô còn có thể dẫn tới dp[k1][k2]: j >= k - (số phần tử còn lại)
        con = n - 1 - i
        lo1, hi1 = max(0, k1 - con), min(k1, i + 1)
        lo2, hi2 = max(0, k2 - con), min(k2, i + 1)
        s = max(lo2, 1)
        # Đi j1 giảm dần để hàng dp[j1 - 1] vẫn là giá trị cũ
        for j1 in range(hi1, lo1 - 1, -1):
            cur = dp[j1]
            if j1 > 0:
                pre = dp[j1 - 1]
                # Cả đoạn được tính từ giá trị cũ rồi mới gán lại
                cur[s:hi2 + 1] = [
                    max(c, c2 + g2, p + g1, p2 + g12)
                    for c, c2, p, p2 in zip(cur[s:hi2 + 1], cur[s - 1:hi2],
                                            pre[s:hi2 + 1], pre[s - 1:hi2])
                ]
                if lo2 == 0 and pre[0] + g1 > cur[0]:
                    cur[0] = pre[0] + g1
            else:
                cur[s:hi2 + 1] = [
                    max(c, c2 + g2)
                    for c, c2 in zip(cur[s:hi2 + 1], cur[s - 1:hi2])
                ]
    return str(sum(a) - dp[k1][k2])


if __name__ == "__main__":
    try:
        with open("arr.inp") as f:
            data = f.read()
        with open("arr.out", "w") as f:
            f.write(solve_arr(data) + "\n")
    except FileNotFoundError:
        print(solve_arr(sys.stdin.read()))
    arr.cpp
    #include <algorithm>
#include <cstdio>
#include <iostream>
#include <vector>
using namespace std;

long long solve_arr(long long b, int k1, int k2, const vector<long long> &a) {
    int n = a.size();
    k1 = min(k1, n);
    k2 = min(k2, n);
    // dp[j1][j2] = lượng giảm lớn nhất khi dùng tối đa j1 phép loại 1, j2 phép loại 2
    vector<vector<long long>> dp(k1 + 1, vector<long long>(k2 + 1, 0));
    long long tong = 0;
    for (int i = 0; i < n; i++) {
        long long x = a[i];
        tong += x;
        long long v1 = x / 2;                                  // chỉ loại 1
        long long v2 = max(x - b, 0LL);                        // chỉ loại 2
        long long v12 = min(max(v1 - b, 0LL), v2 / 2);         // cả hai, chọn thứ tự tốt hơn
        long long g1 = x - v1, g2 = x - v2, g12 = x - v12;
        // Đi ngược cả hai chiều để dp[j1-1][*] và dp[j1][j2-1] vẫn là giá trị cũ
        for (int j1 = min(k1, i + 1); j1 >= 0; j1--) {
            for (int j2 = min(k2, i + 1); j2 >= 0; j2--) {
                long long v = dp[j1][j2];
                if (j2 > 0) v = max(v, dp[j1][j2 - 1] + g2);
                if (j1 > 0) v = max(v, dp[j1 - 1][j2] + g1);
                if (j1 > 0 && j2 > 0) v = max(v, dp[j1 - 1][j2 - 1] + g12);
                dp[j1][j2] = v;
            }
        }
    }
    return tong - dp[k1][k2];
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    // Có arr.inp thì đọc/ghi file, không thì dùng bàn phím/màn hình
    if (FILE *f = fopen("arr.inp", "r")) {
        fclose(f);
        freopen("arr.inp", "r", stdin);
        freopen("arr.out", "w", stdout);
    }
    int n, k1, k2;
    long long b;
    cin >> n >> b >> k1 >> k2;
    vector<long long> a(n);
    for (long long &x : a) cin >> x;
    cout << solve_arr(b, k1, k2, a) << "\n";
    return 0;
}

    4. Số cặp¶

    Ý tưởng. Cố định đầu trái \(i\), đặt \(f_i(j) = \max(a_i..a_j) - \min(b_i..b_j)\). Khi \(j\) tăng, \(\max a\) không giảm và \(\min b\) không tăng, nên \(f_i(j)\) không giảm theo \(j\). Vậy các \(j\) có \(f_i(j) = 0\) tạo thành một đoạn liên tiếp \([p_i, q_i)\), với:

    • \(p_i\) = vị trí \(j \ge i\) đầu tiên có \(f_i(j) \ge 0\),
    • \(q_i\) = vị trí \(j \ge i\) đầu tiên có \(f_i(j) > 0\) (hoặc \(n\) nếu không có),

    và đáp án là \(\sum_i (q_i - p_i)\).

    Khi \(i\) tăng lên \(i + 1\), đoạn \([i+1, j]\) nằm trong \([i, j]\) nên \(\max a\) không tăng, \(\min b\) không giảm, tức \(f_{i+1}(j) \le f_i(j)\). Do đó \(p_{i+1} \ge p_i\) và \(q_{i+1} \ge q_i\): hai con trỏ chỉ chạy về phía trước. Mỗi con trỏ giữ cửa sổ \([i, p)\) (hoặc \([i, q)\)) bằng hai hàng đợi đơn điệu (deque) để biết ngay \(\max a\) và \(\min b\) của cửa sổ. Mỗi chỉ số vào và ra mỗi deque tối đa một lần: tổng \(O(n)\).

    Cách khác cũng đạt: dựng sparse table cho max \(a\) và min \(b\), rồi với mỗi \(i\) chặt nhị phân tìm \(p_i\), \(q_i\), độ phức tạp \(O(n \log n)\); bản Python của cách này chậm hơn đáng kể.

    Thử trên máy biên soạn với \(n = 10^5\) (giá trị nhỏ nhiều trùng, toàn bằng nhau, dãy tăng, giá trị âm/dương lớn): bản Python khoảng 0,2 giây, bản C++ khoảng 0,1–0,3 giây (phần lớn là đọc dữ liệu).

    Bẫy thường gặp

    • Phần tử có thể âm (\(1 \le |a_i|, |b_i| \le 10^9\)): không khởi tạo max bằng 0.
    • Đáp án có thể tới \(n(n+1)/2 \approx 5 \cdot 10^9\) (ví dụ hai dãy toàn số 7), vượt int.
    • Duyệt mọi cặp \((i, j)\) và cập nhật max/min dần là \(O(n^2)\): chỉ đủ cho ràng buộc 1 và 2.

    Nếu viết bằng C++

    Dùng long long cho biến đếm. Hiệu \(\max a - \min b\) có thể tới \(2 \cdot 10^9\), lời giải lưu \(a\), \(b\) bằng long long để không tràn khi trừ.

    pai.py
    import sys
from collections import deque


class ConTro:
    """Con trỏ e chạy tới vị trí đầu tiên r >= l thỏa max(a[l..r]) - min(b[l..r]) >= nguong.

    Cửa sổ [l, e) được giữ bằng hai hàng đợi đơn điệu: max của a và min của b.
    """

    def __init__(self, a, b, nguong):
        self.a, self.b, self.nguong = a, b, nguong
        self.e = 0
        self.qa = deque()  # chỉ số, a giảm dần
        self.qb = deque()  # chỉ số, b tăng dần

    def tim(self, l):
        a, b, qa, qb = self.a, self.b, self.qa, self.qb
        n = len(a)
        while qa and qa[0] < l:
            qa.popleft()
        while qb and qb[0] < l:
            qb.popleft()
        if self.e < l:
            self.e = l
        e = self.e
        while e < n:
            ma = a[e] if not qa or a[qa[0]] < a[e] else a[qa[0]]
            mb = b[e] if not qb or b[qb[0]] > b[e] else b[qb[0]]
            if ma - mb >= self.nguong:
                break
            # Chưa thỏa: đưa e vào cửa sổ rồi xét e + 1
            while qa and a[qa[-1]] <= a[e]:
                qa.pop()
            qa.append(e)
            while qb and b[qb[-1]] >= b[e]:
                qb.pop()
            qb.append(e)
            e += 1
        self.e = e
        return e


def solve_pai(data):
    t = list(map(int, data.split()))
    n = t[0]
    a = t[1:1 + n]
    b = t[1 + n:1 + 2 * n]
    # Cố định l: f(r) = max a - min b không giảm theo r.
    # p = r đầu tiên có f >= 0, q = r đầu tiên có f > 0 -> có q - p đoạn với f = 0.
    # Khi l tăng, f giảm hoặc giữ nguyên nên p, q chỉ tăng (hai con trỏ).
    cp = ConTro(a, b, 0)
    cq = ConTro(a, b, 1)
    dem = 0
    for l in range(n):
        dem += cq.tim(l) - cp.tim(l)
    return str(dem)


if __name__ == "__main__":
    try:
        with open("pai.inp") as f:
            data = f.read()
        with open("pai.out", "w") as f:
            f.write(solve_pai(data) + "\n")
    except FileNotFoundError:
        print(solve_pai(sys.stdin.read()))

    #include <cstdio>
#include <deque>
#include <iostream>
#include <vector>
using namespace std;

// Con trỏ e chạy tới vị trí đầu tiên r >= l thỏa max(a[l..r]) - min(b[l..r]) >= nguong.
// Cửa sổ [l, e) được giữ bằng hai hàng đợi đơn điệu: max của a và min của b.
struct ConTro {
    const vector<long long> &a, &b;
    long long nguong;
    int e = 0;
    deque<int> qa, qb;  // qa: a giảm dần, qb: b tăng dần

    ConTro(const vector<long long> &a_, const vector<long long> &b_, long long ng)
        : a(a_), b(b_), nguong(ng) {}

    int tim(int l) {
        int n = a.size();
        while (!qa.empty() && qa.front() < l) qa.pop_front();
        while (!qb.empty() && qb.front() < l) qb.pop_front();
        if (e < l) e = l;
        while (e < n) {
            long long ma = (qa.empty() || a[qa.front()] < a[e]) ? a[e] : a[qa.front()];
            long long mb = (qb.empty() || b[qb.front()] > b[e]) ? b[e] : b[qb.front()];
            if (ma - mb >= nguong) break;
            // Chưa thỏa: đưa e vào cửa sổ rồi xét e + 1
            while (!qa.empty() && a[qa.back()] <= a[e]) qa.pop_back();
            qa.push_back(e);
            while (!qb.empty() && b[qb.back()] >= b[e]) qb.pop_back();
            qb.push_back(e);
            e++;
        }
        return e;
    }
};

long long solve_pai(const vector<long long> &a, const vector<long long> &b) {
    int n = a.size();
    // Cố định l: f(r) = max a - min b không giảm theo r.
    // p = r đầu tiên có f >= 0, q = r đầu tiên có f > 0 -> có q - p đoạn với f = 0.
    // Khi l tăng, f giảm hoặc giữ nguyên nên p, q chỉ tăng (hai con trỏ).
    ConTro cp(a, b, 0), cq(a, b, 1);
    long long dem = 0;  // tới n(n+1)/2 ~ 5*10^9 -> long long
    for (int l = 0; l < n; l++) dem += cq.tim(l) - cp.tim(l);
    return dem;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    // Có pai.inp thì đọc/ghi file, không thì dùng bàn phím/màn hình
    if (FILE *f = fopen("pai.inp", "r")) {
        fclose(f);
        freopen("pai.inp", "r", stdin);
        freopen("pai.out", "w", stdout);
    }
    int n;
    cin >> n;
    vector<long long> a(n), b(n);
    for (long long &x : a) cin >> x;
    for (long long &x : b) cin >> x;
    cout << solve_pai(a, b) << "\n";
    return 0;
}
