    Đáp án - Đề 1 Bảng A Tiểu học¶

    Lời giải tham khảo bằng Python 3 và C++ (C++17). Xem lại đề bài.

    STT Bài Ý tưởng Độ phức tạp
    1 Tặng bánh Tham lam + thử phần dư \(O(1)\)
    2 Chia kẹo Chẵn / lẻ \(O(N \log N)\)
    3 Chuỗi lặp đều Đếm tần suất ký tự \(O(\text{len}(S))\)
    4 Tính tích Tổng tiền tố \(O(N)\)

    1. Tặng bánh¶

    Ý tưởng. Giá mỗi cái: hộp lớn \(210/30 = 7\), hộp nhỏ \(38/5 = 7{,}6\), mua lẻ \(9\) nghìn đồng. Hộp lớn rẻ nhất nên mua tối đa \(\lfloor N / 210 \rfloor\) hộp lớn. Phần tiền dư nhỏ hơn 210, nên chỉ cần thử mua \(0..5\) hộp nhỏ, còn lại mua lẻ, rồi lấy kết quả lớn nhất.

    Vì sao không nên bớt hộp lớn: 210 nghìn nếu không mua hộp lớn thì chỉ mua được tối đa 27 cái (5 hộp nhỏ + 2 cái lẻ), ít hơn 30 cái.

    Bẫy thường gặp

    Duyệt hai vòng lặp lồng nhau theo số hộp lớn và số hộp nhỏ cho kết quả đúng, nhưng với \(N = 10^9\) sẽ chạy quá thời gian.

    hinchchunhat.py
    def solve_tang_banh():
    n = int(input().strip())

    # Hộp lớn rẻ nhất (7 nghìn/cái) nên mua tối đa hộp lớn,
    # phần tiền dư (< 210) thử từng số hộp nhỏ, còn lại mua lẻ.
    box_large = n // 210
    remaining = n - box_large * 210
    ans = 0

    for box_small in range(remaining // 38 + 1):
        left = remaining - box_small * 38
        cakes = box_large * 30 + box_small * 5 + left // 9
        if cakes > ans:
            ans = cakes

    print(ans)


if __name__ == "__main__":
    solve_tang_banh()
    hinchchunhat.cpp
    #include <iostream>
using namespace std;

void solve_tang_banh() {
    long long n;
    cin >> n;

    // Hộp lớn rẻ nhất (7 nghìn/cái) nên mua tối đa hộp lớn,
    // phần tiền dư (< 210) thử từng số hộp nhỏ, còn lại mua lẻ.
    long long box_large = n / 210;
    long long remaining = n - box_large * 210;
    long long ans = 0;

    for (long long box_small = 0; box_small <= remaining / 38; box_small++) {
        long long left = remaining - box_small * 38;
        long long cakes = box_large * 30 + box_small * 5 + left / 9;
        if (cakes > ans) ans = cakes;
    }

    cout << ans << "\n";
}

int main() {
    solve_tang_banh();
    return 0;
}

    2. Chia kẹo¶

    Ý tưởng. Hai số liên tiếp luôn có một số chẵn và một số lẻ. Vì vậy 2 hộp luôn đủ: hộp số chẵn và hộp số lẻ.

    • Nếu trong dãy có hai số màu liên tiếp → cần 2 hộp.
    • Ngược lại → 1 hộp.

    Sắp xếp dãy rồi kiểm tra hai phần tử kề nhau có chênh lệch đúng 1 hay không.

    bangcuuchuong.py
    def solve_chia_keo():
    n = int(input().strip())
    arr = list(map(int, input().split()))
    arr.sort()

    for i in range(1, n):
        if arr[i] - arr[i - 1] == 1:
            print(2)
            return

    print(1)


if __name__ == "__main__":
    solve_chia_keo()
    bangcuuchuong.cpp
    #include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

void solve_chia_keo() {
    int n;
    cin >> n;
    vector<long long> arr(n);
    for (auto &x : arr) cin >> x;
    sort(arr.begin(), arr.end());

    for (int i = 1; i < n; i++) {
        if (arr[i] - arr[i - 1] == 1) {
            cout << 2 << "\n";
            return;
        }
    }

    cout << 1 << "\n";
}

int main() {
    solve_chia_keo();
    return 0;
}

    3. Chuỗi lặp đều¶

    Ý tưởng. Chuỗi kết quả gồm \(k\) bản sao của một chuỗi con \(T\), nên mỗi ký tự phải xuất hiện số lần chia hết cho \(k\).

    • Nếu có ký tự nào có số lần xuất hiện không chia hết cho \(k\) → in -1.
    • Ngược lại, \(T\) gồm mỗi ký tự c lặp lại \(\text{cnt}(c)/k\) lần; in \(T\) lặp \(k\) lần.

    Đề cho phép in một chuỗi thỏa mãn bất kỳ; lời giải dưới đây in chuỗi có \(T\) đã sắp xếp theo thứ tự chữ cái.

    santruong.py
    def solve_chuoi_lap_deu():
    k = int(input().strip())
    s = input().strip()

    if len(s) % k != 0:
        print(-1)
        return

    cnt = {}
    for ch in s:
        cnt[ch] = cnt.get(ch, 0) + 1

    base = []
    for ch in sorted(cnt):
        if cnt[ch] % k != 0:
            print(-1)
            return
        base.extend([ch] * (cnt[ch] // k))

    result = ''.join(base) * k
    print(result)


if __name__ == "__main__":
    solve_chuoi_lap_deu()
    santruong.cpp
    #include <iostream>
#include <string>
using namespace std;

void solve_chuoi_lap_deu() {
    int k;
    string s;
    cin >> k >> s;

    if (s.size() % k != 0) {
        cout << -1 << "\n";
        return;
    }

    // Đếm số lần xuất hiện của từng ký tự
    int cnt[256] = {0};
    for (unsigned char ch : s) cnt[ch]++;

    string base;
    for (int ch = 0; ch < 256; ch++) {  // duyệt theo thứ tự chữ cái
        if (cnt[ch] == 0) continue;
        if (cnt[ch] % k != 0) {
            cout << -1 << "\n";
            return;
        }
        base += string(cnt[ch] / k, (char)ch);
    }

    string result;
    for (int i = 0; i < k; i++) result += base;
    cout << result << "\n";
}

int main() {
    solve_chuoi_lap_deu();
    return 0;
}

    4. Tính tích¶

    Ý tưởng. Gọi \(T\) là tổng cả dãy. Khi cắt sau phần tử thứ \(i\), phần trái có tổng \(P_i\) (tổng tiền tố) và phần phải có tổng \(T - P_i\). Duyệt mọi vị trí cắt \(i = 1..N-1\) và lấy giá trị lớn nhất của \(P_i \times (T - P_i)\).

    Nếu viết bằng C++/Pascal

    Tích có thể lên tới khoảng \((5 \cdot 10^7)^2 = 2{,}5 \cdot 10^{15}\), vượt kiểu int 32-bit. Lời giải C++ dùng long long; Pascal dùng int64.

    dongho.py
    def solve_tinh_tich():
    n = int(input().strip())
    arr = list(map(int, input().split()))

    total = sum(arr)
    prefix = 0
    ans = 0

    for i in range(n - 1):
        prefix += arr[i]
        ans = max(ans, prefix * (total - prefix))

    print(ans)


if __name__ == "__main__":
    solve_tinh_tich()
    dongho.cpp
    #include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

void solve_tinh_tich() {
    int n;
    cin >> n;
    vector<long long> arr(n);
    for (auto &x : arr) cin >> x;

    long long total = 0;
    for (long long x : arr) total += x;
    long long prefix = 0;
    long long ans = 0;  // tích tới ~2,5e15 nên phải dùng long long

    for (int i = 0; i < n - 1; i++) {
        prefix += arr[i];
        ans = max(ans, prefix * (total - prefix));
    }

    cout << ans << "\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    solve_tinh_tich();
    return 0;
}
