    Đáp án - Bảng A Quận Ngũ Hành Sơn 2025¶

    Lời giải tham khảo bằng Python 3 và C++ (C++17); riêng Bài 1 (vẽ bằng turtle) chỉ có bản Python. Xem lại đề bài.

    Đề thi chính thức làm bằng Scratch 3.0. Code Python dưới đây giúp hiểu thuật toán; khi làm trên Scratch, mỗi hàm Python tương ứng một khối tự tạo, mỗi list tương ứng một Danh sách.

    STT Bài Ý tưởng Lời giải Python
    1 Họa tiết đan lưới Khối tự tạo vẽ đa giác đều, xoay quanh tâm Minh họa bằng turtle
    2 Xử lý danh sách Đảo chữ số + sắp xếp chèn Có, \(O(N^2)\)
    3 Trò chơi 2D Va chạm theo khoảng cách tọa độ Không (bài thiết kế game trên Scratch)

    1. Vẽ hình học & họa tiết đan lưới¶

    Ý tưởng.

    • Khối đa giác (bán kính, số cạnh): từ tâm, nhấc bút đi ra một đoạn bằng bán kính, đặt bút và vẽ đa giác đều. Cạnh đa giác nội tiếp có độ dài \(2R \sin(\pi / n)\), mỗi lần quay \(360 / n\) độ.
    • Khối họa tiết (bán kính, số cánh hoa): lặp số cánh hoa lần, mỗi lần vẽ một đa giác rồi quay \(360 / \text{số cánh hoa}\) độ quanh tâm.
    • Gọi khối họa tiết với nhiều bán kính khác nhau để tạo lưới đan.

    Bẫy thường gặp

    Quên nhấc bút khi đưa bút về tâm hoặc quên trả lại hướng ban đầu sau mỗi đa giác. Hình sau sẽ bị lệch hoặc có nét thừa. Trên Scratch, lưu x, y, hướng vào biến trước khi vẽ và khôi phục sau khi vẽ.

    Bài này không có đáp án duy nhất; code dưới đây chỉ là một cách minh họa bằng thư viện turtle (chạy trên máy tính, mở cửa sổ vẽ).

    hoatiet.py
    import math
import turtle


def da_giac(t, ban_kinh, so_canh):
    # Vẽ đa giác đều nội tiếp đường tròn bán kính ban_kinh, tâm là vị trí hiện tại
    x, y = t.position()
    huong = t.heading()
    t.penup()
    t.forward(ban_kinh)
    t.left(90 + 180 / so_canh)
    t.pendown()
    canh = 2 * ban_kinh * math.sin(math.pi / so_canh)
    for _ in range(so_canh):
        t.forward(canh)
        t.left(360 / so_canh)
    # Nhấc bút, đưa về tâm và hướng cũ để hình sau không bị lem nét
    t.penup()
    t.goto(x, y)
    t.setheading(huong)


def hoa_tiet(t, ban_kinh, so_canh_hoa):
    # "Khối tự tạo" nhận 2 tham số: bán kính và số cánh hoa
    for _ in range(so_canh_hoa):
        da_giac(t, ban_kinh, 6)
        t.left(360 / so_canh_hoa)


def main():
    t = turtle.Turtle()
    t.speed(0)
    t.hideturtle()
    t.penup()
    t.goto(0, 0)
    for ban_kinh in range(40, 161, 40):
        hoa_tiet(t, ban_kinh, 12)
    turtle.done()


if __name__ == "__main__":
    main()

    2. Xử lý danh sách & số học¶

    Ý tưởng.

    1. Kiểm tra số đối xứng bằng cách đảo chữ số: lặp lấy x mod 10 ghép vào số đảo, rồi chia nguyên x cho 10. Số đối xứng khi số đảo bằng chính nó. Trên Scratch có thể dùng khối chữ cái thứ i của để so sánh từ hai đầu.
    2. Danh sách phụ: duyệt dãy, số nào đối xứng thì thêm vào danh sách đối xứng.
    3. Sắp xếp chèn: với \(N \le 100\), sắp xếp chèn hoặc sắp xếp chọn là đủ nhanh và dễ dựng bằng khối lặp lại cho đến khi.
    4. Tổng số lẻ: duyệt danh sách đã lọc, cộng các số có x mod 2 = 1.

    Cách hiểu đề

    Lời giải theo cách hiểu đã ghi ở đề bài: tổng các số lẻ trong danh sách số đối xứng. Nếu ban tổ chức yêu cầu tổng các số lẻ của cả dãy, chỉ cần đổi vòng cộng sang duyệt arr.

    Nếu viết bằng C++

    Đề không giới hạn độ lớn các số. Lời giải C++ đọc số bằng long long (tới khoảng \(9{,}2 \cdot 10^{18}\)), đảo chữ số bằng unsigned long long để không tràn, và cộng tổng số lẻ bằng __int128 vì 100 số cỡ \(10^{18}\) có tổng vượt long long.

    doixung.py
    def dao_nguoc(x):
    # Đảo chữ số bằng phép chia lấy dư - cách làm giống trên Scratch
    r = 0
    while x > 0:
        r = r * 10 + x % 10
        x //= 10
    return r


def solve_so_doi_xung():
    n = int(input().strip())
    arr = list(map(int, input().split()))

    # Danh sách phụ: chỉ giữ các số đối xứng
    doi_xung = []
    for x in arr:
        if dao_nguoc(x) == x:
            doi_xung.append(x)

    # Sắp xếp chèn: đơn giản, dễ chuyển sang Scratch
    for i in range(1, len(doi_xung)):
        x = doi_xung[i]
        j = i - 1
        while j >= 0 and doi_xung[j] > x:
            doi_xung[j + 1] = doi_xung[j]
            j -= 1
        doi_xung[j + 1] = x

    tong_le = 0
    for x in doi_xung:
        if x % 2 == 1:
            tong_le += x

    if doi_xung:
        print(*doi_xung)
    else:
        print(-1)
    print(tong_le)


if __name__ == "__main__":
    solve_so_doi_xung()
    doixung.cpp
    #include <iostream>
#include <string>
#include <vector>
using namespace std;

unsigned long long dao_nguoc(unsigned long long x) {
    // Đảo chữ số bằng phép chia lấy dư - cách làm giống trên Scratch
    // (unsigned long long: số đảo của số 19 chữ số vẫn không tràn)
    unsigned long long r = 0;
    while (x > 0) {
        r = r * 10 + x % 10;
        x /= 10;
    }
    return r;
}

// In số __int128 (cout không in trực tiếp được)
void in_so(__int128 v) {
    if (v == 0) { cout << 0; return; }
    string s;
    while (v > 0) { s += char('0' + v % 10); v /= 10; }
    for (int i = (int)s.size() - 1; i >= 0; i--) cout << s[i];
}

void solve_so_doi_xung() {
    int n;
    cin >> n;
    vector<long long> arr(n);
    for (auto &x : arr) cin >> x;

    // Danh sách phụ: chỉ giữ các số đối xứng
    vector<long long> doi_xung;
    for (long long x : arr) {
        if (dao_nguoc(x) == (unsigned long long)x) doi_xung.push_back(x);
    }

    // Sắp xếp chèn: đơn giản, dễ chuyển sang Scratch
    for (int i = 1; i < (int)doi_xung.size(); i++) {
        long long x = doi_xung[i];
        int j = i - 1;
        while (j >= 0 && doi_xung[j] > x) {
            doi_xung[j + 1] = doi_xung[j];
            j--;
        }
        doi_xung[j + 1] = x;
    }

    // Tổng có thể vượt long long (100 số cỡ 1e18) nên dùng __int128
    __int128 tong_le = 0;
    for (long long x : doi_xung) {
        if (x % 2 == 1) tong_le += x;
    }

    if (!doi_xung.empty()) {
        for (int i = 0; i < (int)doi_xung.size(); i++) {
            if (i > 0) cout << " ";
            cout << doi_xung[i];
        }
        cout << "\n";
    } else {
        cout << -1 << "\n";
    }
    in_so(tong_le);
    cout << "\n";
}

int main() {
    solve_so_doi_xung();
    return 0;
}

    3. Mô phỏng trò chơi tương tác 2D¶

    Bài thiết kế game trên Scratch, chấm theo sản phẩm nên không có đáp án Python. Gợi ý các điểm kỹ thuật:

    • Vật rơi: dùng bản sao (clone), xuất hiện ở x ngẫu nhiên trên đỉnh sân khấu, mỗi khung hình giảm y theo biến tốc độ.
    • Va chạm theo tọa độ: coi là chạm khi \(\sqrt{(x_1 - x_2)^2 + (y_1 - y_2)^2} < r_1 + r_2\) (hoặc dùng khối khoảng cách đến).
    • Level: khi điểm đạt mốc, tăng tốc độ hoặc giảm thời gian giữa hai lần tạo vật rơi.
    • Thời gian: dùng đồng hồ bấm giờ có sẵn, hết giờ thì phát tín hiệu thắng/thua.
