    Đáp án - Bảng A Quận Sơn Trà 2022¶

    Lời giải tham khảo bằng Python 3 và C++ (C++17). Xem lại đề bài.

    Đề gốc nhập từ bàn phím và in ra màn hình. Mỗi chương trình dưới đây đọc input.inp và ghi output.out nếu có file input.inp; nếu không có thì đọc từ bàn phím và in ra màn hình như đề yêu cầu.

    STT Bài Ý tưởng Độ phức tạp
    1 Bóng đèn Chu vi chia khoảng cách: \(8a\) \(O(1)\)
    2 Biểu thức nhỏ nhất \(x\) nhỏ nhất, \(y\) lớn nhất: \(a - d\) \(O(1)\)
    3 Không thích số 3 Dãy lặp lại theo chu kỳ 30 (18 số mỗi chu kỳ) \(O(1)\)
    4 Số cân bằng Đếm theo số chữ số của nửa trái \(O(\log n)\)

    1. Bóng đèn¶

    Ý tưởng. Đổi đơn vị: cạnh \(a\) dm \(= 10a\) cm, chu vi hình vuông \(= 40a\) cm. Các bóng xếp thành một vòng khép kín, cứ 5 cm một bóng, nên số bóng bằng số khoảng: \(40a / 5 = 8a\).

    Kiểm tra: \(a = 1\) → 8 bóng; \(a = 3\) → 24 bóng.

    Bẫy thường gặp

    • Quên đổi dm sang cm.
    • Cộng thêm 1 bóng như bài trồng cây trên đoạn thẳng. Đây là đường khép kín nên số bóng bằng số khoảng.
    bongden.py
    import sys


def solve_bongden(data):
    a = int(data.split()[0])
    # Chu vi bảng: 4 * a dm = 40 * a cm, cứ 5 cm một bóng
    so_bong = 40 * a // 5
    return str(so_bong)


if __name__ == "__main__":
    try:
        with open("input.inp") as f:
            data = f.read()
        with open("output.out", "w") as f:
            f.write(solve_bongden(data) + "\n")
    except FileNotFoundError:
        print(solve_bongden(sys.stdin.read()))
    bongden.cpp
    #include <cstdio>
#include <iostream>
using namespace std;

long long solve_bongden(long long a) {
    // Chu vi bảng: 4 * a dm = 40 * a cm, cứ 5 cm một bóng
    long long so_bong = 40 * a / 5;
    return so_bong;
}

int main() {
    if (FILE *f = fopen("input.inp", "r")) {
        fclose(f);
        freopen("input.inp", "r", stdin);
        freopen("output.out", "w", stdout);
    }
    long long a;
    cin >> a;
    cout << solve_bongden(a) << "\n";
    return 0;
}

    2. Biểu thức nhỏ nhất¶

    Ý tưởng. \(x - y\) nhỏ nhất khi \(x\) nhỏ nhất và \(y\) lớn nhất, tức \(x = a\), \(y = d\). Đáp án là \(a - d\).

    Ví dụ: \(a = 10, d = 7\) → \(3\).

    Bẫy thường gặp

    • Duyệt mọi cặp \((x, y)\): với giá trị tới \(10^{12}\) sẽ chạy mãi không xong.
    • Kết quả có thể âm (ví dụ \(a = 1\), \(d = 5\) → \(-4\)).

    Nếu viết bằng C++

    Giá trị tới \(10^{12}\) nên lời giải dùng long long.

    bieuthuc.py
    import sys


def solve_bieuthuc(data):
    a, b, c, d = map(int, data.split()[:4])
    # x - y nhỏ nhất khi x nhỏ nhất (x = a) và y lớn nhất (y = d)
    return str(a - d)


if __name__ == "__main__":
    try:
        with open("input.inp") as f:
            data = f.read()
        with open("output.out", "w") as f:
            f.write(solve_bieuthuc(data) + "\n")
    except FileNotFoundError:
        print(solve_bieuthuc(sys.stdin.read()))
    bieuthuc.cpp
    #include <cstdio>
#include <iostream>
using namespace std;

// b, c không ảnh hưởng tới kết quả
long long solve_bieuthuc(long long a, long long /*b*/, long long /*c*/, long long d) {
    // x - y nhỏ nhất khi x nhỏ nhất (x = a) và y lớn nhất (y = d)
    return a - d;
}

int main() {
    if (FILE *f = fopen("input.inp", "r")) {
        fclose(f);
        freopen("input.inp", "r", stdin);
        freopen("output.out", "w", stdout);
    }
    long long a, b, c, d;
    cin >> a >> b >> c >> d;
    cout << solve_bieuthuc(a, b, c, d) << "\n";
    return 0;
}

    3. Không thích số 3¶

    Ý tưởng. Một số có được thích hay không chỉ phụ thuộc vào số dư khi chia cho 3 và chữ số tận cùng (số dư khi chia cho 10), nên dãy lặp lại theo chu kỳ \(30\). Trong các số từ 1 đến 30 có 18 số được thích:

    1, 2, 4, 5, 7, 8, 10, 11, 14, 16, 17, 19, 20, 22, 25, 26, 28, 29.

    Số thứ \(k\) là \(30 \cdot \lfloor (k-1)/18 \rfloor + \text{mau}[(k-1) \bmod 18]\) (đánh số danh sách mau từ 0).

    Ví dụ: \(k = 9\) → nhóm 0, vị trí 8 → \(14\). \(k = 20\) → nhóm 1, vị trí 1 → \(30 + 2 = 32\).

    Bẫy thường gặp

    • Duyệt từng số cho đến khi đếm đủ \(k\) số: với \(k = 10^9\) phải duyệt khoảng \(1{,}67 \cdot 10^9\) số, quá chậm (đặc biệt trong Scratch). Cách duyệt chỉ đủ cho \(k\) nhỏ.
    • Ví dụ 2 trong đề gốc ghi Output 17 là sai, đáp án đúng là 14.
    khongthich.py
    import sys


def solve_khongthich(data):
    k = int(data.split()[0])
    # Trong mỗi đoạn 30 số liên tiếp có đúng 18 số Polycarp thích
    mau = []
    for x in range(1, 31):
        if x % 3 != 0 and x % 10 != 3:
            mau.append(x)
    # mau = [1, 2, 4, 5, 7, 8, 10, 11, 14, 16, 17, 19, 20, 22, 25, 26, 28, 29]
    nhom = (k - 1) // 18
    vi_tri = (k - 1) % 18
    return str(30 * nhom + mau[vi_tri])


if __name__ == "__main__":
    try:
        with open("input.inp") as f:
            data = f.read()
        with open("output.out", "w") as f:
            f.write(solve_khongthich(data) + "\n")
    except FileNotFoundError:
        print(solve_khongthich(sys.stdin.read()))
    khongthich.cpp
    #include <cstdio>
#include <iostream>
#include <vector>
using namespace std;

long long solve_khongthich(long long k) {
    // Trong mỗi đoạn 30 số liên tiếp có đúng 18 số Polycarp thích
    vector<long long> mau;
    for (int x = 1; x <= 30; x++) {
        if (x % 3 != 0 && x % 10 != 3) {
            mau.push_back(x);
        }
    }
    // mau = {1, 2, 4, 5, 7, 8, 10, 11, 14, 16, 17, 19, 20, 22, 25, 26, 28, 29}
    long long nhom = (k - 1) / 18;
    long long vi_tri = (k - 1) % 18;
    return 30 * nhom + mau[vi_tri];
}

int main() {
    if (FILE *f = fopen("input.inp", "r")) {
        fclose(f);
        freopen("input.inp", "r", stdin);
        freopen("output.out", "w", stdout);
    }
    long long k;
    cin >> k;
    cout << solve_khongthich(k) << "\n";
    return 0;
}

    4. Số cân bằng¶

    Ý tưởng. Số cân bằng có \(2L\) chữ số là số viết nửa trái \(x\) (\(x\) có đúng \(L\) chữ số) hai lần liền nhau, tức bằng \(x \cdot (10^L + 1)\). Ví dụ \(12 \cdot 101 = 1212\).

    Với mỗi \(L = 1, 2, \dots\): \(x\) chạy từ \(10^{L-1}\) đến \(\min(10^L - 1, \lfloor n / (10^L + 1) \rfloor)\); cộng số giá trị đó vào kết quả. Dừng khi số cân bằng nhỏ nhất có \(2L\) chữ số đã vượt \(n\). Với \(n \le 10^{12}\) chỉ cần \(L \le 6\).

    Ví dụ \(n = 1333\): \(L = 1\) cho 9 số (11..99); \(L = 2\): \(\lfloor 1333/101 \rfloor = 13\) → \(x = 10..13\), thêm 4 số. Tổng 13.

    Bẫy thường gặp

    • Duyệt mọi số từ 1 đến \(n\) và kiểm tra: với \(n = 10^{12}\) là quá chậm.
    • Một cách vừa đủ nhanh và dễ viết trong Scratch: cho \(x\) chạy \(1, 2, 3, \dots\), ghép \(x\) với chính nó thành số cân bằng, dừng khi số đó vượt \(n\) (tối đa khoảng \(10^6\) lần lặp).

    Nếu viết bằng C++

    \(n\) tới \(10^{12}\) nên lời giải dùng long long. Điều kiện dừng viết dạng nho > n / he_so thay cho nho * he_so > n để phép nhân không tràn số.

    socanbang.py
    import sys


def solve_socanbang(data):
    n = int(data.split()[0])
    dem = 0
    # Số cân bằng có 2L chữ số = x * (10^L + 1), với x có đúng L chữ số
    nho = 1          # 10^(L-1): số nhỏ nhất có L chữ số
    while True:
        lon = nho * 10 - 1          # số lớn nhất có L chữ số
        he_so = nho * 10 + 1        # 10^L + 1
        if nho * he_so > n:         # số cân bằng nhỏ nhất đã vượt n
            break
        x_max = min(lon, n // he_so)
        dem += x_max - nho + 1
        nho = nho * 10
    return str(dem)


if __name__ == "__main__":
    try:
        with open("input.inp") as f:
            data = f.read()
        with open("output.out", "w") as f:
            f.write(solve_socanbang(data) + "\n")
    except FileNotFoundError:
        print(solve_socanbang(sys.stdin.read()))
    socanbang.cpp
    #include <algorithm>
#include <cstdio>
#include <iostream>
using namespace std;

long long solve_socanbang(long long n) {
    long long dem = 0;
    // Số cân bằng có 2L chữ số = x * (10^L + 1), với x có đúng L chữ số
    long long nho = 1;                  // 10^(L-1): số nhỏ nhất có L chữ số
    while (true) {
        long long lon = nho * 10 - 1;   // số lớn nhất có L chữ số
        long long he_so = nho * 10 + 1; // 10^L + 1
        // nho * he_so > n, viết dạng chia để không tràn số
        if (nho > n / he_so) {          // số cân bằng nhỏ nhất đã vượt n
            break;
        }
        long long x_max = min(lon, n / he_so);
        dem += x_max - nho + 1;
        nho = nho * 10;
    }
    return dem;
}

int main() {
    if (FILE *f = fopen("input.inp", "r")) {
        fclose(f);
        freopen("input.inp", "r", stdin);
        freopen("output.out", "w", stdout);
    }
    long long n;
    cin >> n;
    cout << solve_socanbang(n) << "\n";
    return 0;
}
